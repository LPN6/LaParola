1 וְשָׁם נִקְרָא אִישׁ בְּלִיַּעַל וּשְׁמֹו שֶׁבַע בֶּנ־בִּכְרִי אִישׁ יְמִינִי וַיִּתְקַע בַּשֹּׁפָר וַיֹּאמֶר אֵינ־לָנוּ חֵלֶק בְּדָוִד וְלֹא נַחֲלָה־לָנוּ בְּבֶנ־יִשַׁי אִישׁ לְאֹהָלָיו יִשְׂרָאֵל׃ 2 וַיַּעַל כָּל־אִישׁ יִשְׂרָאֵל מֵאַחֲרֵי דָוִד אַחֲרֵי שֶׁבַע בֶּנ־בִּכְרִי וְאִישׁ יְהוּדָה דָּבְקוּ בְמַלְכָּם מִנ־הַיַּרְדֵּן וְעַד־יְרוּשָׁלִָם׃ 3 וַיָּבֹא דָוִד אֶל־בֵּיתֹו יְרוּשָׁלִַם וַיִּקַּח הַמֶּלֶךְ אֵת עֶשֶׂר־נָשִׁים ׀ פִּלַגְשִׁים אֲשֶׁר הִנִּיחַ לִשְׁמֹר הַבַּיִת וַיִּתְּנֵם בֵּית־מִשְׁמֶרֶת וַיְכַלְכְּלֵם וַאֲלֵיהֶם לֹא־בָא וַתִּהְיֶינָה צְרֻרֹות עַד־יֹום מֻתָן אַלְמְנוּת חַיּוּת׃ ס 4 וַיֹּאמֶר הַמֶּלֶךְ אֶל־עֲמָשָׂא הַזְעֶק־לִי אֶת־אִישׁ־יְהוּדָה שְׁלֹשֶׁת יָמִים וְאַתָּה פֹּה עֲמֹד׃ 5 וַיֵּלֶךְ עֲמָשָׂא לְהַזְעִיק אֶת־יְהוּדָה וַיִּיחֶר וַיֹּוחֶר מִנ־הַמֹּועֵד אֲשֶׁר יְעָדֹו׃ ס 6 וַיֹּאמֶר דָּוִד אֶל־אֲבִישַׁי עַתָּה יֵרַע לָנוּ שֶׁבַע בֶּנ־בִּכְרִי מִנ־אַבְשָׁלֹום אַתָּה קַח אֶת־עַבְדֵי אֲדֹנֶיךָ וּרְדֹף אַחֲרָיו פֶּנ־מָצָא לֹו עָרִים בְּצֻרֹות וְהִצִּיל עֵינֵנוּ׃ 7 וַיֵּצְאוּ אַחֲרָיו אַנְשֵׁי יֹואָב וְהַכְּרֵתִי וְהַפְּלֵתִי וְכָל־הַגִּבֹּרִים וַיֵּצְאוּ מִירוּשָׁלִַם לִרְדֹּף אַחֲרֵי שֶׁבַע בֶּנ־בִּכְרִי׃ 8 הֵם עִמ־הָאֶבֶן הַגְּדֹולָה אֲשֶׁר בְּגִבְעֹון וַעֲמָשָׂא בָּא לִפְנֵיהֶם וְיֹואָב חָגוּר ׀ מִדֹּו לְבֻשׁוּ וְעָלֹו וְעָלָיו חֲגֹור חֶרֶב מְצֻמֶּדֶת עַל־מָתְנָיו בְּתַעְרָהּ וְהוּא יָצָא וַתִּפֹּל׃ ס 9 וַיֹּאמֶר יֹואָב לַעֲמָשָׂא הֲשָׁלֹום אַתָּה אָחִי וַתֹּחֶז יַד־יְמִין יֹואָב בִּזְקַן עֲמָשָׂא לִנְשָׁק־לֹו׃ 10 וַעֲמָשָׂא לֹא־נִשְׁמַר בַּחֶרֶב ׀ אֲשֶׁר בְּיַד־יֹואָב וַיַּכֵּהוּ בָהּ אֶל־הַחֹמֶשׁ וַיִּשְׁפֹּךְ מֵעָיו אַרְצָה וְלֹא־שָׁנָה לֹו וַיָּמֹת ס וְיֹואָב וַאֲבִישַׁי אָחִיו רָדַף אַחֲרֵי שֶׁבַע בֶּנ־בִּכְרִי׃ 11 וְאִישׁ עָמַד עָלָיו מִנַּעֲרֵי יֹואָב וַיֹּאמֶר מִי אֲשֶׁר חָפֵץ בְּיֹואָב וּמִי אֲשֶׁר־לְדָוִד אַחֲרֵי יֹואָב׃ 12 וַעֲמָשָׂא מִתְגֹּלֵל בַּדָּם בְּתֹוךְ הַמְּסִלָּה וַיַּרְא הָאִישׁ כִּי־עָמַד כָּל־הָעָם וַיַּסֵּב אֶת־עֲמָשָׂא מִנ־הַמְסִלָּה הַשָּׂדֶה וַיַּשְׁלֵךְ עָלָיו בֶּגֶד כַּאֲשֶׁר רָאָה כָּל־הַבָּא עָלָיו וְעָמָד׃ 13 כַּאֲשֶׁר הֹגָה מִנ־הַמְסִלָּה עָבַר כָּל־אִישׁ אַחֲרֵי יֹואָב לִרְדֹּף אַחֲרֵי שֶׁבַע בֶּנ־בִּכְרִי׃ 14 וַיַּעֲבֹר בְּכָל־שִׁבְטֵי יִשְׂרָאֵל אָבֵלָה וּבֵית מַעֲכָה וְכָל־הַבֵּרִים ס וַיִּקְלֵהוּ וַיִּקָּהֲלוּ וַיָּבֹאוּ אַפ־אַחֲרָיו׃ 15 וַיָּבֹאוּ וַיָּצֻרוּ עָלָיו בְּאָבֵלָה בֵּית הַמַּעֲכָה וַיִּשְׁפְּכוּ סֹלְלָה אֶל־הָעִיר וַתַּעֲמֹד בַּחֵל וְכָל־הָעָם אֲשֶׁר אֶת־יֹואָב מַשְׁחִיתִם לְהַפִּיל הַחֹומָה׃ 16 וַתִּקְרָא אִשָּׁה חֲכָמָה מִנ־הָעִיר שִׁמְעוּ שִׁמְעוּ אִמְרוּ־נָא אֶל־יֹואָב קְרַב עַד־הֵנָּה וַאֲדַבְּרָה אֵלֶיךָ׃ 17 וַיִּקְרַב אֵלֶיהָ וַתֹּאמֶר הָאִשָּׁה הַאַתָּה יֹואָב וַיֹּאמֶר אָנִי וַתֹּאמֶר לֹו שְׁמַע דִּבְרֵי אֲמָתֶךָ וַיֹּאמֶר שֹׁמֵעַ אָנֹכִי׃ 18 וַתֹּאמֶר לֵאמֹר דַּבֵּר יְדַבְּרוּ בָרִאשֹׁנָה לֵאמֹר שָׁאֹל יְשָׁאֲלוּ בְּאָבֵל וְכֵן הֵתַמּוּ׃ 19 אָנֹכִי שְׁלֻמֵי אֱמוּנֵי יִשְׂרָאֵל אַתָּה מְבַקֵּשׁ לְהָמִית עִיר וְאֵם בְּיִשְׂרָאֵל לָמָּה תְבַלַּע נַחֲלַת יְהוָה׃ ף 20 וַיַּעַן יֹואָב וַיֹּאמַר חָלִילָה חָלִילָה לִי אִמ־אֲבַלַּע וְאִמ־אַשְׁחִית׃ 21 לֹא־כֵן הַדָּבָר כִּי אִישׁ מֵהַר אֶפְרַיִם שֶׁבַע בֶּנ־בִּכְרִי שְׁמֹו נָשָׂא יָדֹו בַּמֶּלֶךְ בְּדָוִד תְּנוּ־אֹתֹו לְבַדֹּו וְאֵלְכָה מֵעַל הָעִיר וַתֹּאמֶר הָאִשָּׁה אֶל־יֹואָב הִנֵּה רֹאשֹׁו מֻשְׁלָךְ אֵלֶיךָ בְּעַד הַחֹומָה׃ 22 וַתָּבֹוא הָאִשָּׁה אֶל־כָּל־הָעָם בְּחָכְמָתָהּ וַיִּכְרְתוּ אֶת־רֹאשׁ שֶׁבַע בֶּנ־בִּכְרִי וַיַּשְׁלִכוּ אֶל־יֹואָב וַיִּתְקַע בַּשֹּׁופָר וַיָּפֻצוּ מֵעַל־הָעִיר אִישׁ לְאֹהָלָיו וְיֹואָב שָׁב יְרוּשָׁלִַם אֶל־הַמֶּלֶךְ׃ ס 23 וְיֹואָב אֶל כָּל־הַצָּבָא יִשְׂרָאֵל וּבְנָיָה בֶּנ־יְהֹויָדָע עַל־הַכְּרֵי הַכְּרֵתִי וְעַל־הַפְּלֵתִי׃
