1 בִּשְׁנַת עֶשְׂרִים וָשֶׁבַע שָׁנָה לְיָרָבְעָם מֶלֶךְ יִשְׂרָאֵל מָלַךְ עֲזַרְיָה בֶנ־אֲמַצְיָה מֶלֶךְ יְהוּדָה׃ 2 בֶּנ־שֵׁשׁ עֶשְׂרֵה שָׁנָה הָיָה בְמָלְכֹו וַחֲמִשִּׁים וּשְׁתַּיִם שָׁנָה מָלַךְ בִּירוּשָׁלִָם וְשֵׁם אִמֹּו יְכָלְיָהוּ מִירוּשָׁלִָם׃ 3 וַיַּעַשׂ הַיָּשָׁר בְּעֵינֵי יְהוָה כְּכֹל אֲשֶׁר־עָשָׂה אֲמַצְיָהוּ אָבִיו׃ 4 רַק הַבָּמֹות לֹא־סָרוּ עֹוד הָעָם מְזַבְּחִים וּמְקַטְּרִים בַּבָּמֹות׃ 5 וַיְנַגַּע יְהוָה אֶת־הַמֶּלֶךְ וַיְהִי מְצֹרָע עַד־יֹום מֹתֹו וַיֵּשֶׁב בְּבֵית הַחָפְשִׁית וְיֹותָם בֶּנ־הַמֶּלֶךְ עַל־הַבַּיִת שֹׁפֵט אֶת־עַם הָאָרֶץ׃ 6 וְיֶתֶר דִּבְרֵי עֲזַרְיָהוּ וְכָל־אֲשֶׁר עָשָׂה הֲלֹא־הֵם כְּתוּבִים עַל־סֵפֶר דִּבְרֵי הַיָּמִים לְמַלְכֵי יְהוּדָה׃ 7 וַיִּשְׁכַּב עֲזַרְיָה עִמ־אֲבֹתָיו וַיִּקְבְּרוּ אֹתֹו עִמ־אֲבֹתָיו בְּעִיר דָּוִד וַיִּמְלֹךְ יֹותָם בְּנֹו תַּחְתָּיו׃ ף 8 בִּשְׁנַת שְׁלֹשִׁים וּשְׁמֹנֶה שָׁנָה לַעֲזַרְיָהוּ מֶלֶךְ יְהוּדָה מָלַךְ זְכַרְיָהוּ בֶנ־יָרָבְעָם עַל־יִשְׂרָאֵל בְּשֹׁמְרֹון שִׁשָּׁה חֳדָשִׁים׃ 9 וַיַּעַשׂ הָרַע בְּעֵינֵי יְהוָה כַּאֲשֶׁר עָשׂוּ אֲבֹתָיו לֹא סָר מֵחַטֹּאות יָרָבְעָם בֶּנ־נְבָט אֲשֶׁר הֶחֱטִיא אֶת־יִשְׂרָאֵל׃ 10 וַיִּקְשֹׁר עָלָיו שַׁלֻּם בֶּנ־יָבֵשׁ וַיַּכֵּהוּ קָבָלְ עָם וַיְמִיתֵהוּ וַיִּמְלֹךְ תַּחְתָּיו׃ 11 וְיֶתֶר דִּבְרֵי זְכַרְיָה הִנָּם כְּתוּבִים עַל־סֵפֶר דִּבְרֵי הַיָּמִים לְמַלְכֵי יִשְׂרָאֵל׃ 12 הוּא דְבַר־יְהוָה אֲשֶׁר דִּבֶּר אֶל־יֵהוּא לֵאמֹר בְּנֵי רְבִיעִים יֵשְׁבוּ לְךָ עַל־כִּסֵּא יִשְׂרָאֵל וַיְהִי־כֵן׃ ף 13 שַׁלּוּם בֶּנ־יָבֵישׁ מָלַךְ בִּשְׁנַת שְׁלֹשִׁים וָתֵשַׁע שָׁנָה לְעֻזִיָּה מֶלֶךְ יְהוּדָה וַיִּמְלֹךְ יֶרַח־יָמִים בְּשֹׁמְרֹון׃ 14 וַיַּעַל מְנַחֵם בֶּנ־גָּדִי מִתִּרְצָה וַיָּבֹא שֹׁמְרֹון וַיַּךְ אֶת־שַׁלּוּם בֶּנ־יָבֵישׁ בְּשֹׁמְרֹון וַיְמִיתֵהוּ וַיִּמְלֹךְ תַּחְתָּיו׃ 15 וְיֶתֶר דִּבְרֵי שַׁלּוּם וְקִשְׁרֹו אֲשֶׁר קָשָׁר הִנָּם כְּתֻבִים עַל־סֵפֶר דִּבְרֵי הַיָּמִים לְמַלְכֵי יִשְׂרָאֵל׃ ס 16 אָז יַכֶּה־מְנַחֵם אֶת־תִּפְסַח וְאֶת־כָּל־אֲשֶׁר־בָּהּ וְאֶת־גְּבוּלֶיהָ מִתִּרְצָה כִּי לֹא פָתַח וַיַּךְ אֵת כָּל־הֶהָרֹותֶיהָ בִּקֵּעַ׃ ף 17 בִּשְׁנַת שְׁלֹשִׁים וָתֵשַׁע שָׁנָה לַעֲזַרְיָה מֶלֶךְ יְהוּדָה מָלַךְ מְנַחֵם בֶּנ־גָּדִי עַל־יִשְׂרָאֵל עֶשֶׂר שָׁנִים בְּשֹׁמְרֹון׃ 18 וַיַּעַשׂ הָרַע בְּעֵינֵי יְהוָה לֹא סָר מֵעַל חַטֹּאות יָרָבְעָם בֶּנ־נְבָט אֲשֶׁר־הֶחֱטִיא אֶת־יִשְׂרָאֵל כָּל־יָמָיו׃ 19 בָּא פוּל מֶלֶכְ־אַשּׁוּר עַל־הָאָרֶץ וַיִּתֵּן מְנַחֵם לְפוּל אֶלֶף כִּכַּר־כָּסֶף לִהְיֹות יָדָיו אִתֹּו לְהַחֲזִיק הַמַּמְלָכָה בְּיָדֹו׃ 20 וַיֹּצֵא מְנַחֵם אֶת־הַכֶּסֶף עַל־יִשְׂרָאֵל עַל כָּל־גִּבֹּורֵי הַחַיִל לָתֵת לְמֶלֶךְ אַשּׁוּר חֲמִשִּׁים שְׁקָלִים כֶּסֶף לְאִישׁ אֶחָד וַיָּשָׁב מֶלֶךְ אַשּׁוּר וְלֹא־עָמַד שָׁם בָּאָרֶץ׃ 21 וְיֶתֶר דִּבְרֵי מְנַחֵם וְכָל־אֲשֶׁר עָשָׂה הֲלֹוא־הֵם כְּתוּבִים עַל־סֵפֶר דִּבְרֵי הַיָּמִים לְמַלְכֵי יִשְׂרָאֵל׃ 22 וַיִּשְׁכַּב מְנַחֵם עִמ־אֲבֹתָיו וַיִּמְלֹךְ פְּקַחְיָה בְנֹו תַּחְתָּיו׃ ף 23 בִּשְׁנַת חֲמִשִּׁים שָׁנָה לַעֲזַרְיָה מֶלֶךְ יְהוּדָה מָלַךְ פְּקַחְיָה בֶנ־מְנַחֵם עַל־יִשְׂרָאֵל בְּשֹׁמְרֹון שְׁנָתָיִם׃ 24 וַיַּעַשׂ הָרַע בְּעֵינֵי יְהוָה לֹא סָר מֵחַטֹּאות יָרָבְעָם בֶּנ־נְבָט אֲשֶׁר הֶחֱטִיא אֶת־יִשְׂרָאֵל׃ 25 וַיִּקְשֹׁר עָלָיו פֶּקַח בֶּנ־רְמַלְיָהוּ שָׁלִישֹׁו וַיַּכֵּהוּ בְשֹׁמְרֹון בְּאַרְמֹון בֵּית־מֶלֶךְ הַמֶּלֶךְ אֶת־אַרְגֹּב וְאֶת־הָאַרְיֵה וְעִמֹּו חֲמִשִּׁים אִישׁ מִבְּנֵי גִלְעָדִים וַיְמִיתֵהוּ וַיִּמְלֹךְ תַּחְתָּיו׃ 26 וְיֶתֶר דִּבְרֵי פְקַחְיָה וְכָל־אֲשֶׁר עָשָׂה הִנָּם כְּתוּבִים עַל־סֵפֶר דִּבְרֵי הַיָּמִים לְמַלְכֵי יִשְׂרָאֵל׃ ף 27 בִּשְׁנַת חֲמִשִּׁים וּשְׁתַּיִם שָׁנָה לַעֲזַרְיָה מֶלֶךְ יְהוּדָה מָלַךְ פֶּקַח בֶּנ־רְמַלְיָהוּ עַל־יִשְׂרָאֵל בְּשֹׁמְרֹון עֶשְׂרִים שָׁנָה׃ 28 וַיַּעַשׂ הָרַע בְּעֵינֵי יְהוָה לֹא סָר מִנ־חַטֹּאות יָרָבְעָם בֶּנ־נְבָט אֲשֶׁר הֶחֱטִיא אֶת־יִשְׂרָאֵל׃ 29 בִּימֵי פֶּקַח מֶלֶכְ־יִשְׂרָאֵל בָּא תִּגְלַת פִּלְאֶסֶר מֶלֶךְ אַשּׁוּר וַיִּקַּח אֶת־עִיֹּון וְאֶת־אָבֵל בֵּית־מַעֲכָה וְאֶת־יָנֹוחַ וְאֶת־קֶדֶשׁ וְאֶת־חָצֹור וְאֶת־הַגִּלְעָד וְאֶת־הַגָּלִילָה כֹּל אֶרֶץ נַפְתָּלִי וַיַּגְלֵם אַשּׁוּרָה׃ 30 וַיִּקְשָׁר־קֶשֶׁר הֹושֵׁעַ בֶּנ־אֵלָה עַל־פֶּקַח בֶּנ־רְמַלְיָהוּ וַיַּכֵּהוּ וַיְמִיתֵהוּ וַיִּמְלֹךְ תַּחְתָּיו בִּשְׁנַת עֶשְׂרִים לְיֹותָם בֶּנ־עֻזִיָּה׃ 31 וְיֶתֶר דִּבְרֵי־פֶקַח וְכָל־אֲשֶׁר עָשָׂה הִנָּם כְּתוּבִים עַל־סֵפֶר דִּבְרֵי הַיָּמִים לְמַלְכֵי יִשְׂרָאֵל׃ ף 32 בִּשְׁנַת שְׁתַּיִם לְפֶקַח בֶּנ־רְמַלְיָהוּ מֶלֶךְ יִשְׂרָאֵל מָלַךְ יֹותָם בֶּנ־עֻזִיָּהוּ מֶלֶךְ יְהוּדָה׃ 33 בֶּנ־עֶשְׂרִים וְחָמֵשׁ שָׁנָה הָיָה בְמָלְכֹו וְשֵׁשׁ־עֶשְׂרֵה שָׁנָה מָלַךְ בִּירוּשָׁלִָם וְשֵׁם אִמֹּו יְרוּשָׁא בַּת־צָדֹוק׃ 34 וַיַּעַשׂ הַיָּשָׁר בְּעֵינֵי יְהוָה כְּכֹל אֲשֶׁר־עָשָׂה עֻזִיָּהוּ אָבִיו עָשָׂה׃ 35 רַק הַבָּמֹות לֹא סָרוּ עֹוד הָעָם מְזַבְּחִים וּמְקַטְּרִים בַּבָּמֹות הוּא בָּנָה אֶת־שַׁעַר בֵּית־יְהוָה הָעֶלְיֹון׃ 36 וְיֶתֶר דִּבְרֵי יֹותָם אֲשֶׁר עָשָׂה הֲלֹא־הֵם כְּתוּבִים עַל־סֵפֶר דִּבְרֵי הַיָּמִים לְמַלְכֵי יְהוּדָה׃ 37 בַּיָּמִים הָהֵם הֵחֵל יְהוָה לְהַשְׁלִיחַ בִּיהוּדָה רְצִין מֶלֶךְ אֲרָם וְאֵת פֶּקַח בֶּנ־רְמַלְיָהוּ׃ 38 וַיִּשְׁכַּב יֹותָם עִמ־אֲבֹתָיו וַיִּקָּבֵר עִמ־אֲבֹתָיו בְּעִיר דָּוִד אָבִיו וַיִּמְלֹךְ אָחָז
