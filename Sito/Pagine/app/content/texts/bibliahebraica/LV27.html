1 וַיְדַבֵּר יְהוָה אֶל־מֹשֶׁה לֵּאמֹר׃ 2 דַּבֵּר אֶל־בְּנֵי יִשְׂרָאֵל וְאָמַרְתָּ אֲלֵהֶם אִישׁ כִּי יַפְלִא נֶדֶר בְּעֶרְכְּךָ נְפָשֹׁת לַיהוָה׃ 3 וְהָיָה עֶרְכְּךָ הַזָּכָר מִבֶּן עֶשְׂרִים שָׁנָה וְעַד בֶּנ־שִׁשִּׁים שָׁנָה וְהָיָה עֶרְכְּךָ חֲמִשִּׁים שֶׁקֶל כֶּסֶף בְּשֶׁקֶל הַקֹּדֶשׁ׃ 4 וְאִמ־נְקֵבָה הִוא וְהָיָה עֶרְכְּךָ שְׁלֹשִׁים שָׁקֶל׃ 5 וְאִם מִבֶּנ־חָמֵשׁ שָׁנִים וְעַד בֶּנ־עֶשְׂרִים שָׁנָה וְהָיָה עֶרְכְּךָ הַזָּכָר עֶשְׂרִים שְׁקָלִים וְלַנְּקֵבָה עֲשֶׂרֶת שְׁקָלִים׃ 6 וְאִם מִבֶּנ־חֹדֶשׁ וְעַד בֶּנ־חָמֵשׁ שָׁנִים וְהָיָה עֶרְכְּךָ הַזָּכָר חֲמִשָּׁה שְׁקָלִים כָּסֶף וְלַנְּקֵבָה עֶרְכְּךָ שְׁלֹשֶׁת שְׁקָלִים כָּסֶף׃ 7 וְאִם מִבֶּנ־שִׁשִּׁים שָׁנָה וָמַעְלָה אִמ־זָכָר וְהָיָה עֶרְכְּךָ חֲמִשָּׁה עָשָׂר שָׁקֶל וְלַנְּקֵבָה עֲשָׂרָה שְׁקָלִים׃ 8 וְאִמ־מָךְ הוּא מֵעֶרְכֶּךָ וְהֶעֱמִידֹו לִפְנֵי הַכֹּהֵן וְהֶעֱרִיךְ אֹתֹו הַכֹּהֵן עַל־פִּי אֲשֶׁר תַּשִּׂיג יַד הַנֹּדֵר יַעֲרִיכֶנּוּ הַכֹּהֵן׃ ס 9 וְאִמ־בְּהֵמָה אֲשֶׁר יַקְרִיבוּ מִמֶּנָּה קָרְבָּן לַיהוָה כֹּל אֲשֶׁר יִתֵּן מִמֶּנּוּ לַיהוָה יִהְיֶה־קֹּדֶשׁ׃ 10 לֹא יַחֲלִיפֶנּוּ וְלֹא־יָמִיר אֹתֹו טֹוב בְּרָע אֹו־רַע בְּטֹוב וְאִמ־הָמֵר יָמִיר בְּהֵמָה בִּבְהֵמָה וְהָיָה־הוּא וּתְמוּרָתֹו יִהְיֶה־קֹּדֶשׁ׃ 11 וְאִם כָּל־בְּהֵמָה טְמֵאָה אֲשֶׁר לֹא־יַקְרִיבוּ מִמֶּנָּה קָרְבָּן לַיהוָה וְהֶעֱמִיד אֶת־הַבְּהֵמָה לִפְנֵי הַכֹּהֵן׃ 12 וְהֶעֱרִיךְ הַכֹּהֵן אֹתָהּ בֵּין טֹוב וּבֵין רָע כְּעֶרְכְּךָ הַכֹּהֵן כֵּן יִהְיֶה׃ 13 וְאִמ־גָּאֹל יִגְאָלֶנָּה וְיָסַף חֲמִישִׁתֹו עַל־עֶרְכֶּךָ׃ 14 וְאִישׁ כִּי־יַקְדִּשׁ אֶת־בֵּיתֹו קֹדֶשׁ לַיהוָה וְהֶעֱרִיכֹו הַכֹּהֵן בֵּין טֹוב וּבֵין רָע כַּאֲשֶׁר יַעֲרִיךְ אֹתֹו הַכֹּהֵן כֵּן יָקוּם׃ 15 וְאִמ־הַמַּקְדִּישׁ יִגְאַל אֶת־בֵּיתֹו וְיָסַף חֲמִישִׁית כֶּסֶפ־עֶרְכְּךָ עָלָיו וְהָיָה לֹו׃ 16 וְאִם ׀ מִשְּׂדֵה אֲחֻזָּתֹו יַקְדִּישׁ אִישׁ לַיהוָה וְהָיָה עֶרְכְּךָ לְפִי זַרְעֹו זֶרַע חֹמֶר שְׂעֹרִים בַּחֲמִשִּׁים שֶׁקֶל כָּסֶף׃ 17 אִמ־מִשְּׁנַת הַיֹּבֵל יַקְדִּישׁ שָׂדֵהוּ כְּעֶרְכְּךָ יָקוּם׃ 18 וְאִמ־אַחַר הַיֹּבֵל יַקְדִּישׁ שָׂדֵהוּ וְחִשַּׁב־לֹו הַכֹּהֵן אֶת־הַכֶּסֶף עַל־פִּי הַשָּׁנִים הַנֹּותָרֹת עַד שְׁנַת הַיֹּבֵל וְנִגְרַע מֵעֶרְכֶּךָ׃ 19 וְאִמ־גָּאֹל יִגְאַל אֶת־הַשָּׂדֶה הַמַּקְדִּישׁ אֹתֹו וְיָסַף חֲמִשִׁית כֶּסֶפ־עֶרְכְּךָ עָלָיו וְקָם לֹו׃ 20 וְאִמ־לֹא יִגְאַל אֶת־הַשָּׂדֶה וְאִמ־מָכַר אֶת־הַשָּׂדֶה לְאִישׁ אַחֵר לֹא יִגָּאֵל עֹוד׃ 21 וְהָיָה הַשָּׂדֶה בְּצֵאתֹו בַיֹּבֵל קֹדֶשׁ לַיהוָה כִּשְׂדֵה הַחֵרֶם לַכֹּהֵן תִּהְיֶה אֲחֻזָּתֹו׃ 22 וְאִם אֶת־שְׂדֵה מִקְנָתֹו אֲשֶׁר לֹא מִשְּׂדֵה אֲחֻזָּתֹו יַקְדִּישׁ לַיהוָה׃ 23 וְחִשַּׁב־לֹו הַכֹּהֵן אֵת מִכְסַת הָעֶרְכְּךָ עַד שְׁנַת הַיֹּבֵל וְנָתַן אֶת־הָעֶרְכְּךָ בַּיֹּום הַהוּא קֹדֶשׁ לַיהוָה׃ 24 בִּשְׁנַת הַיֹּובֵל יָשׁוּב הַשָּׂדֶה לַאֲשֶׁר קָנָהוּ מֵאִתֹּו לַאֲשֶׁר־לֹו אֲחֻזַּת הָאָרֶץ׃ 25 וְכָל־עֶרְכְּךָ יִהְיֶה בְּשֶׁקֶל הַקֹּדֶשׁ עֶשְׂרִים גֵּרָה יִהְיֶה הַשָּׁקֶל׃ ס 26 אַכְ־בְּכֹור אֲשֶׁר־יְבֻכַּר לַיהוָה בִּבְהֵמָה לֹא־יַקְדִּישׁ אִישׁ אֹתֹו אִמ־שֹׁור אִמ־שֶׂה לַיהוָה הוּא׃ 27 וְאִם בַּבְּהֵמָה הַטְּמֵאָה וּפָדָה בְעֶרְכֶּךָ וְיָסַף חֲמִשִׁתֹו עָלָיו וְאִמ־לֹא יִגָּאֵל וְנִמְכַּר בְּעֶרְכֶּךָ׃ 28 אַכְ־כָּל־חֵרֶם אֲשֶׁר יַחֲרִם אִישׁ לַיהוָה מִכָּל־אֲשֶׁר־לֹו מֵאָדָם וּבְהֵמָה וּמִשְּׂדֵה אֲחֻזָּתֹו לֹא יִמָּכֵר וְלֹא יִגָּאֵל כָּל־חֵרֶם קֹדֶשׁ־קָדָשִׁים הוּא לַיהוָה׃ 29 כָּל־חֵרֶם אֲשֶׁר יָחֳרַם מִנ־הָאָדָם לֹא יִפָּדֶה מֹות יוּמָת׃ 30 וְכָל־מַעְשַׂר הָאָרֶץ מִזֶּרַע הָאָרֶץ מִפְּרִי הָעֵץ לַיהוָה הוּא קֹדֶשׁ לַיהוָה׃ 31 וְאִמ־גָּאֹל יִגְאַל אִישׁ מִמַּעַשְׂרֹו חֲמִשִׁיתֹו יֹסֵף עָלָיו׃ 32 וְכָל־מַעְשַׂר בָּקָר וָצֹאן כֹּל אֲשֶׁר־יַעֲבֹר תַּחַת הַשָּׁבֶט הָעֲשִׂירִי יִהְיֶה־קֹּדֶשׁ לַיהוָה׃ 33 לֹא יְבַקֵּר בֵּינ־טֹוב לָרַע וְלֹא יְמִירֶנּוּ וְאִמ־הָמֵר יְמִירֶנּוּ וְהָיָה־הוּא וּתְמוּרָתֹו יִהְיֶה־קֹדֶשׁ לֹא יִגָּאֵל׃ 34 אֵלֶּה הַמִּצְוֹת אֲשֶׁר צִוָּה יְהוָה אֶת־מֹשֶׁה אֶל־בְּנֵי יִשְׂרָאֵל בְּהַר סִינָי׃
