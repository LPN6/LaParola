18
1 וַיְהִי לִיהֹושָׁפָט עֹשֶׁר וְכָבֹוד לָרֹב וַיִּתְחַתֵּן לְאַחְאָב׃ 2 וַיֵּרֶד לְקֵץ שָׁנִים אֶל־אַחְאָב לְשֹׁמְרֹון וַיִּזְבַּח־לֹו אַחְאָב צֹאן וּבָקָר לָרֹב וְלָעָם אֲשֶׁר עִמֹּו וַיְסִיתֵהוּ לַעֲלֹות אֶל־רָמֹות גִּלְעָד׃ 3 וַיֹּאמֶר אַחְאָב מֶלֶכְ־יִשְׂרָאֵל אֶל־יְהֹושָׁפָט מֶלֶךְ יְהוּדָה הֲתֵלֵךְ עִמִּי רָמֹת גִּלְעָד וַיֹּאמֶר לֹו כָּמֹונִי כָמֹוךָ וּכְעַמְּךָ עַמִּי וְעִמְּךָ בַּמִּלְחָמָה׃ 4 וַיֹּאמֶר יְהֹושָׁפָט אֶל־מֶלֶךְ יִשְׂרָאֵל דְּרָשׁ־נָא כַיֹּום אֶת־דְּבַר יְהוָה׃ 5 וַיִּקְבֹּץ מֶלֶכְ־יִשְׂרָאֵל אֶת־הַנְּבִאִים אַרְבַּע מֵאֹות אִישׁ וַיֹּאמֶר אֲלֵהֶם הֲנֵלֵךְ אֶל־רָמֹת גִּלְעָד לַמִּלְחָמָה אִמ־אֶחְדָּל וַיֹּאמְרוּ עֲלֵה וְיִתֵּן הָאֱלֹהִים בְּיַד הַמֶּלֶךְ׃ 6 וַיֹּאמֶר יְהֹושָׁפָט הַאֵין פֹּה נָבִיא לַיהוָה עֹוד וְנִדְרְשָׁה מֵאֹתֹו׃ 7 וַיֹּאמֶר מֶלֶכְ־יִשְׂרָאֵל ׀ אֶל־יְהֹושָׁפָט עֹוד אִישׁ־אֶחָד לִדְרֹושׁ אֶת־יְהוָה מֵאֹתֹו וַאֲנִי שְׂנֵאתִיהוּ כִּי־אֵינֶנּוּ מִתְנַבֵּא עָלַי לְטֹובָה כִּי כָל־יָמָיו לְרָעָה הוּא מִיכָיְהוּ בֶנ־יִמְלָא וַיֹּאמֶר יְהֹושָׁפָט אַל־יֹאמַר הַמֶּלֶךְ כֵּן׃ 8 וַיִּקְרָא מֶלֶךְ יִשְׂרָאֵל אֶל־סָרִיס אֶחָד וַיֹּאמֶר מַהֵר מִיכָהוּ מִיכָיְהוּ בֶנ־יִמְלָא׃ 9 וּמֶלֶךְ יִשְׂרָאֵל וִיהֹושָׁפָט מֶלֶכְ־יְהוּדָה יֹושְׁבִים אִישׁ עַל־כִּסְאֹו מְלֻבָּשִׁים בְּגָדִים וְיֹשְׁבִים בְּגֹרֶן פֶּתַח שַׁעַר שֹׁמְרֹון וְכָל־הַנְּבִיאִים מִתְנַבְּאִים לִפְנֵיהֶם׃ 10 וַיַּעַשׂ לֹו צִדְקִיָּהוּ בֶנ־כְּנַעֲנָה קַרְנֵי בַרְזֶל וַיֹּאמֶר כֹּה־אָמַר יְהוָה בְּאֵלֶּה תְּנַגַּח אֶת־אֲרָם עַד־כַּלֹּותָם׃ 11 וְכָל־הַנְּבִאִים נִבְּאִים כֵּן לֵאמֹר עֲלֵה רָמֹת גִּלְעָד וְהַצְלַח וְנָתַן יְהוָה בְּיַד הַמֶּלֶךְ׃ 12 וְהַמַּלְאָךְ אֲשֶׁר־הָלַךְ ׀ לִקְרֹא לְמִיכָיְהוּ דִּבֶּר אֵלָיו לֵאמֹר הִנֵּה דִּבְרֵי הַנְּבִאִים פֶּה־אֶחָד טֹוב אֶל־הַמֶּלֶךְ וִיהִי־נָא דְבָרְךָ כְּאַחַד מֵהֶם וְדִבַּרְתָּ טֹּוב׃ 13 וַיֹּאמֶר מִיכָיְהוּ חַי־יְהוָה כִּי אֶת־אֲשֶׁר־יֹאמַר אֱלֹהַי אֹתֹו אֲדַבֵּר׃ 14 וַיָּבֹא אֶל־הַמֶּלֶךְ וַיֹּאמֶר הַמֶּלֶךְ אֵלָיו מִיכָה הֲנֵלֵךְ אֶל־רָמֹת גִּלְעָד לַמִּלְחָמָה אִמ־אֶחְדָּל וַיֹּאמֶר עֲלוּ וְהַצְלִיחוּ וְיִנָּתְנוּ בְּיֶדְכֶם׃ 15 וַיֹּאמֶר אֵלָיו הַמֶּלֶךְ עַד־כַּמֶּה פְעָמִים אֲנִי מַשְׁבִּיעֶךָ אֲשֶׁר לֹא־תְדַבֵּר אֵלַי רַק־אֱמֶת בְּשֵׁם יְהוָה׃ 16 וַיֹּאמֶר רָאִיתִי אֶת־כָּל־יִשְׂרָאֵל נְפֹוצִים עַל־הֶהָרִים כַּצֹּאן אֲשֶׁר אֵינ־לָהֶן רֹעֶה וַיֹּאמֶר יְהוָה לֹא־אֲדֹנִים לָאֵלֶּה יָשׁוּבוּ אִישׁ־לְבֵיתֹו בְּשָׁלֹום׃ 17 וַיֹּאמֶר מֶלֶכְ־יִשְׂרָאֵל אֶל־יְהֹושָׁפָט הֲלֹא אָמַרְתִּי אֵלֶיךָ לֹא־יִתְנַבֵּא עָלַי טֹוב כִּי אִמ־לְרָע׃ ס 18 וַיֹּאמֶר לָכֵן שִׁמְעוּ דְבַר־יְהוָה רָאִיתִי אֶת־יְהוָה יֹושֵׁב עַל־כִּסְאֹו וְכָל־צְבָא הַשָּׁמַיִם עֹמְדִים עַל־יְמִינֹו וּשְׂמֹאלֹו׃ 19 וַיֹּאמֶר יְהוָה מִי יְפַתֶּה אֶת־אַחְאָב מֶלֶכְ־יִשְׂרָאֵל וְיַעַל וְיִפֹּל בְּרָמֹות גִּלְעָד וַיֹּאמֶר זֶה אֹמֵר כָּכָה וְזֶה אֹמֵר כָּכָה׃ 20 וַיֵּצֵא הָרוּחַ וַיַּעֲמֹד לִפְנֵי יְהוָה וַיֹּאמֶר אֲנִי אֲפַתֶּנּוּ וַיֹּאמֶר יְהוָה אֵלָיו בַּמָּה׃ 21 וַיֹּאמֶר אֵצֵא וְהָיִיתִי לְרוּחַ שֶׁקֶר בְּפִי כָּל־נְבִיאָיו וַיֹּאמֶר תְּפַתֶּה וְגַמ־תּוּכָל צֵא וַעֲשֵׂה־כֵן׃ 22 וְעַתָּה הִנֵּה נָתַן יְהוָה רוּחַ שֶׁקֶר בְּפִי נְבִיאֶיךָ אֵלֶּה וַיהוָה דִּבֶּר עָלֶיךָ רָעָה׃ ס 23 וַיִּגַּשׁ צִדְקִיָּהוּ בֶנ־כְּנַעֲנָה וַיַּךְ אֶת־מִיכָיְהוּ עַל־הַלֶּחִי וַיֹּאמֶר אֵי זֶה הַדֶּרֶךְ עָבַר רוּחַ־יְהוָה מֵאִתִּי לְדַבֵּר אֹתָךְ׃ 24 וַיֹּאמֶר מִיכָיְהוּ הִנְּךָ רֹאֶה בַּיֹּום הַהוּא אֲשֶׁר תָּבֹוא חֶדֶר בְּחֶדֶר לְהֵחָבֵא׃ 25 וַיֹּאמֶר מֶלֶךְ יִשְׂרָאֵל קְחוּ אֶת־מִיכָיְהוּ וַהֲשִׁיבֻהוּ אֶל־אָמֹון שַׂר־הָעִיר וְאֶל־יֹואָשׁ בֶּנ־הַמֶּלֶךְ׃ 26 וַאֲמַרְתֶּם כֹּה אָמַר הַמֶּלֶךְ שִׂימוּ זֶה בֵּית הַכֶּלֶא וְהַאֲכִלֻהוּ לֶחֶם לַחַץ וּמַיִם לַחַץ עַד שׁוּבִי בְשָׁלֹום׃ 27 וַיֹּאמֶר מִיכָיְהוּ אִמ־שֹׁוב תָּשׁוּב בְּשָׁלֹום לֹא־דִבֶּר יְהוָה בִּי וַיֹּאמֶר שִׁמְעוּ עַמִּים כֻּלָּם׃ ף 28 וַיַּעַל מֶלֶכְ־יִשְׂרָאֵל וִיהֹושָׁפָט מֶלֶכְ־יְהוּדָה אֶל־רָמֹת גִּלְעָד׃ 29 וַיֹּאמֶר מֶלֶךְ יִשְׂרָאֵל אֶל־יְהֹושָׁפָט הִתְחַפֵּשׂ וָבֹוא בַמִּלְחָמָה וְאַתָּה לְבַשׁ בְּגָדֶיךָ וַיִּתְחַפֵּשׂ מֶלֶךְ יִשְׂרָאֵל וַיָּבֹאוּ בַּמִּלְחָמָה׃ 30 וּמֶלֶךְ אֲרָם צִוָּה אֶת־שָׂרֵי הָרֶכֶב אֲשֶׁר־לֹו לֵאמֹר לֹא תִּלָּחֲמוּ אֶת־הַקָּטֹן אֶת־הַגָּדֹול כִּי אִמ־אֶת־מֶלֶךְ יִשְׂרָאֵל לְבַדֹּו׃ 31 וַיְהִי כִּרְאֹות שָׂרֵי הָרֶכֶב אֶת־יְהֹושָׁפָט וְהֵמָּה אָמְרוּ מֶלֶךְ יִשְׂרָאֵל הוּא וַיָּסֹבּוּ עָלָיו לְהִלָּחֵם וַיִּזְעַק יְהֹושָׁפָט וַיהוָה עֲזָרֹו וַיְסִיתֵם אֱלֹהִים מִמֶּנּוּ׃ 32 וַיְהִי כִּרְאֹות שָׂרֵי הָרֶכֶב כִּי לֹא־הָיָה מֶלֶךְ יִשְׂרָאֵל וַיָּשֻׁבוּ מֵאַחֲרָיו׃ 33 וְאִישׁ מָשַׁךְ בַּקֶּשֶׁת לְתֻמֹּו וַיַּךְ אֶת־מֶלֶךְ יִשְׂרָאֵל בֵּין הַדְּבָקִים וּבֵין הַשִּׁרְיָן וַיֹּאמֶר לָרַכָּב הֲפֹךְ יָדֶיךָ יָדְךָ וְהֹוצֵאתַנִי מִנ־הַמַּחֲנֶה כִּי הָחֳלֵיתִי׃ 34 וַתַּעַל הַמִּלְחָמָה בַּיֹּום הַהוּא וּמֶלֶךְ יִשְׂרָאֵל הָיָה מַעֲמִיד בַּמֶּרְכָּבָה נֹכַח אֲרָם עַד־הָעָרֶב וַיָּמָת לְעֵת בֹּוא הַשָּׁמֶשׁ׃
