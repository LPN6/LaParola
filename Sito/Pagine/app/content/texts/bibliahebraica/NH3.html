1 וַיָּקָם אֶלְיָשִׁיב הַכֹּהֵן הַגָּדֹול וְאֶחָיו הַכֹּהֲנִים וַיִּבְנוּ אֶת־שַׁעַר הַצֹּאן הֵמָּה קִדְּשׁוּהוּ וַיַּעֲמִידוּ דַּלְתֹתָיו וְעַד־מִגְדַּל הַמֵּאָה קִדְּשׁוּהוּ עַד מִגְדַּל חֲנַנְאֵל׃ ס 2 וְעַל־יָדֹו בָנוּ אַנְשֵׁי יְרֵחֹו ס וְעַל־יָדֹו בָנָה זַכּוּר בֶּנ־אִמְרִי׃ ס 3 וְאֵת שַׁעַר הַדָּגִים בָּנוּ בְּנֵי הַסְּנָאָה הֵמָּה קֵרוּהוּ וַיַּעֲמִידוּ דַּלְתֹתָיו מַנְעוּלָיו וּבְרִיחָיו׃ ס 4 וְעַל־יָדָם הֶחֱזִיק מְרֵמֹות בֶּנ־אוּרִיָּה בֶּנ־הַקֹּוץ ס וְעַל־יָדָם הֶחֱזִיק מְשֻׁלָּם בֶּנ־בֶּרֶכְיָה בֶּנ־מְשֵׁיזַבְאֵל ס וְעַל־יָדָם הֶחֱזִיק צָדֹוק בֶּנ־בַּעֲנָא׃ ס 5 וְעַל־יָדָם הֶחֱזִיקוּ הַתְּקֹועִים וְאַדִּירֵיהֶם לֹא־הֵבִיאוּ צַוָּרָם בַּעֲבֹדַת אֲדֹנֵיהֶם׃ ס 6 וְאֵת שַׁעַר הַיְשָׁנָה הֶחֱזִיקוּ יֹויָדָע בֶּנ־פָּסֵחַ וּמְשֻׁלָּם בֶּנ־בְּסֹודְיָה הֵמָּה קֵרוּהוּ וַיַּעֲמִידוּ דַּלְתֹתָיו וּמַנְעֻלָיו וּבְרִיחָיו׃ ס 7 וְעַל־יָדָם הֶחֱזִיק מְלַטְיָה הַגִּבְעֹנִי וְיָדֹון הַמֵּרֹנֹתִי אַנְשֵׁי גִבְעֹון וְהַמִּצְפָּה לְכִסֵּא פַּחַת עֵבֶר הַנָּהָר׃ ס 8 עַל־יָדֹו הֶחֱזִיק עֻזִּיאֵל בֶּנ־חַרְהֲיָה צֹורְפִים ס וְעַל־יָדֹו הֶחֱזִיק חֲנַנְיָה בֶּנ־הָרַקָּחִים וַיַּעַזְבוּ יְרוּשָׁלִַם עַד הַחֹומָה הָרְחָבָה׃ ס 9 וְעַל־יָדָם הֶחֱזִיק רְפָיָה בֶנ־חוּר שַׂר חֲצִי פֶּלֶךְ יְרוּשָׁלִָם׃ ס 10 וְעַל־יָדָם הֶחֱזִיק יְדָיָה בֶנ־חֲרוּמַף וְנֶגֶד בֵּיתֹו ס וְעַל־יָדֹו הֶחֱזִיק חַטּוּשׁ בֶּנ־חֲשַׁבְנְיָה׃ 11 מִדָּה שֵׁנִית הֶחֱזִיק מַלְכִּיָּה בֶנ־חָרִם וְחַשּׁוּב בֶּנ־פַּחַת מֹואָב וְאֵת מִגְדַּל הַתַּנּוּרִים׃ ס 12 וְעַל־יָדֹו הֶחֱזִיק שַׁלּוּם בֶּנ־הַלֹּוחֵשׁ שַׂר חֲצִי פֶּלֶךְ יְרוּשָׁלִָם הוּא וּבְנֹותָיו׃ ס 13 אֵת שַׁעַר הַגַּיְא הֶחֱזִיק חָנוּן וְיֹשְׁבֵי זָנֹוחַ הֵמָּה בָנוּהוּ וַיַּעֲמִידוּ דַּלְתֹתָיו מַנְעֻלָיו וּבְרִיחָיו וְאֶלֶף אַמָּה בַּחֹומָה עַד שַׁעַר הָשֲׁפֹות׃ 14 וְאֵת ׀ שַׁעַר הָאַשְׁפֹּות הֶחֱזִיק מַלְכִּיָּה בֶנ־רֵכָב שַׂר פֶּלֶךְ בֵּית־הַכָּרֶם הוּא יִבְנֶנּוּ וְיַעֲמִיד דַּלְתֹתָיו מַנְעֻלָיו וּבְרִיחָיו׃ ס 15 וְאֵת שַׁעַר הָעַיִן הֶחֱזִיק שַׁלּוּן בֶּנ־כָּל־חֹזֶה שַׂר פֶּלֶךְ הַמִּצְפָּה הוּא יִבְנֶנּוּ וִיטַלְלֶנּוּ וְיַעֲמִידוּ וְיַעֲמִיד דַּלְתֹתָיו מַנְעֻלָיו וּבְרִיחָיו וְאֵת חֹומַת בְּרֵכַת הַשֶּׁלַח לְגַנ־הַמֶּלֶךְ וְעַד־הַמַּעֲלֹות הַיֹּורְדֹות מֵעִיר דָּוִיד׃ ס 16 אַחֲרָיו הֶחֱזִיק נְחֶמְיָה בֶנ־עַזְבּוּק שַׂר חֲצִי פֶּלֶךְ בֵּית־צוּר עַד־נֶגֶד קִבְרֵי דָוִיד וְעַד־הַבְּרֵכָה הָעֲשׂוּיָה וְעַד בֵּית הַגִּבֹּרִים׃ ס 17 אַחֲרָיו הֶחֱזִיקוּ הַלְוִיִּם רְחוּם בֶּנ־בָּנִי עַל־יָדֹו הֶחֱזִיק חֲשַׁבְיָה שַׂר־חֲצִי־פֶלֶךְ קְעִילָה לְפִלְכֹּו׃ ס 18 אַחֲרָיו הֶחֱזִיקוּ אֲחֵיהֶם בַּוַּי בֶּנ־חֵנָדָד שַׂר חֲצִי פֶּלֶךְ קְעִילָה׃ ס 19 וַיְחַזֵּק עַל־יָדֹו עֵזֶר בֶּנ־יֵשׁוּעַ שַׂר הַמִּצְפָּה מִדָּה שֵׁנִית מִנֶּגֶד עֲלֹת הַנֶּשֶׁק הַמִּקְצֹעַ׃ ס 20 אַחֲרָיו הֶחֱרָה הֶחֱזִיק בָּרוּךְ בֶּנ־זַבַּי זַכַּי מִדָּה שֵׁנִית מִנ־הַמִּקְצֹועַ עַד־פֶּתַח בֵּית אֶלְיָשִׁיב הַכֹּהֵן הַגָּדֹול׃ ס 21 אַחֲרָיו הֶחֱזִיק מְרֵמֹות בֶּנ־אוּרִיָּה בֶּנ־הַקֹּוץ מִדָּה שֵׁנִית מִפֶּתַח בֵּית אֶלְיָשִׁיב וְעַד־תַּכְלִית בֵּית אֶלְיָשִׁיב׃ ס 22 וְאַחֲרָיו הֶחֱזִיקוּ הַכֹּהֲנִים אַנְשֵׁי הַכִּכָּר׃ 23 אַחֲרָיו הֶחֱזִיק בִּנְיָמִן וְחַשּׁוּב נֶגֶד בֵּיתָם ס אַחֲרָיו הֶחֱזִיק עֲזַרְיָה בֶנ־מַעֲשֵׂיָה בֶּנ־עֲנָנְיָה אֵצֶל בֵּיתֹו׃ ס 24 אַחֲרָיו הֶחֱזִיק בִּנּוּי בֶּנ־חֵנָדָד מִדָּה שֵׁנִית מִבֵּית עֲזַרְיָה עַד־הַמִּקְצֹועַ וְעַד־הַפִּנָּה׃ 25 פָּלָל בֶּנ־אוּזַי מִנֶּגֶד הַמִּקְצֹועַ וְהַמִּגְדָּל הַיֹּוצֵא מִבֵּית הַמֶּלֶךְ הָעֶלְיֹון אֲשֶׁר לַחֲצַר הַמַּטָּרָה אַחֲרָיו פְּדָיָה בֶנ־פַּרְעֹשׁ׃ ס 26 וְהַנְּתִינִים הָיוּ יֹשְׁבִים בָּעֹפֶל עַד נֶגֶד שַׁעַר הַמַּיִם לַמִּזְרָח וְהַמִּגְדָּל הַיֹּוצֵא׃ ס 27 אַחֲרָיו הֶחֱזִיקוּ הַתְּקֹעִים מִדָּה שֵׁנִית מִנֶּגֶד הַמִּגְדָּל הַגָּדֹול הַיֹּוצֵא וְעַד חֹומַת הָעֹפֶל׃ 28 מֵעַל ׀ שַׁעַר הַסּוּסִים הֶחֱזִיקוּ הַכֹּהֲנִים אִישׁ לְנֶגֶד בֵּיתֹו׃ ס 29 אַחֲרָיו הֶחֱזִיק צָדֹוק בֶּנ־אִמֵּר נֶגֶד בֵּיתֹו ס וְאַחֲרָיו הֶחֱזִיק שְׁמַעְיָה בֶנ־שְׁכַנְיָה שֹׁמֵר שַׁעַר הַמִּזְרָח׃ ס 30 אַחֲרֵי אַחֲרָיו הֶחֱזִיק חֲנַנְיָה בֶנ־שֶׁלֶמְיָה וְחָנוּן בֶּנ־צָלָף הַשִּׁשִּׁי מִדָּה שֵׁנִי ס אַחֲרָיו הֶחֱזִיק מְשֻׁלָּם בֶּנ־בֶּרֶכְיָה נֶגֶד נִשְׁכָּתֹו׃ ס 31 אַחֲרֵי אַחֲרָיו הֶחֱזִיק מַלְכִּיָּה בֶּנ־הַצֹּרְפִי עַד־בֵּית הַנְּתִינִים וְהָרֹכְלִים נֶגֶד שַׁעַר הַמִּפְקָד וְעַד עֲלִיַּת הַפִּנָּה׃ 32 וּבֵין עֲלִיַּת הַפִּנָּה לְשַׁעַר הַצֹּאן הֶחֱזִיקוּ הַצֹּרְפִים וְהָרֹכְלִים׃ ף 33 וַיְהִי כַּאֲשֶׁר שָׁמַע סַנְבַלַּט כִּי־אֲנַחְנוּ בֹונִים אֶת־הַחֹומָה וַיִּחַר לֹו וַיִּכְעַס הַרְבֵּה וַיַּלְעֵג עַל־הַיְּהוּדִים׃ 34 וַיֹּאמֶר ׀ לִפְנֵי אֶחָיו וְחֵיל שֹׁמְרֹון וַיֹּאמֶר מָה הַיְּהוּדִים הָאֲמֵלָלִים עֹשִׂים הֲיַעַזְבוּ לָהֶם הֲיִזְבָּחוּ הַיְכַלּוּ בַיֹּום הַיְחַיּוּ אֶת־הָאֲבָנִים מֵעֲרֵמֹות הֶעָפָר וְהֵמָּה שְׂרוּפֹות׃ 35 וְטֹובִיָּה הָעַמֹּנִי אֶצְלֹו וַיֹּאמֶר גַּם אֲשֶׁר־הֵם בֹּונִים אִמ־יַעֲלֶה שׁוּעָל וּפָרַץ חֹומַת אַבְנֵיהֶם׃ ף 36 שְׁמַע אֱלֹהֵינוּ כִּי־הָיִינוּ בוּזָה וְהָשֵׁב חֶרְפָּתָם אֶל־רֹאשָׁם וּתְנֵם לְבִזָּה
