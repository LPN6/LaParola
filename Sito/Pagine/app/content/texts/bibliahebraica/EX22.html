1 אִמ־בַּמַּחְתֶּרֶת יִמָּצֵא הַגַּנָּב וְהֻכָּה וָמֵת אֵין לֹו דָּמִים׃ 2 אִמ־זָרְחָה הַשֶּׁמֶשׁ עָלָיו דָּמִים לֹו שַׁלֵּם יְשַׁלֵּם אִמ־אֵין לֹו וְנִמְכַּר בִּגְנֵבָתֹו׃ 3 אִמ־הִמָּצֵא תִמָּצֵא בְיָדֹו הַגְּנֵבָה מִשֹּׁור עַד־חֲמֹור עַד־שֶׂה חַיִּים שְׁנַיִם יְשַׁלֵּם׃ ס 4 כִּי יַבְעֶר־אִישׁ שָׂדֶה אֹו־כֶרֶם וְשִׁלַּח אֶת־בְּעִירָה בְּעִירֹו וּבִעֵר בִּשְׂדֵה אַחֵר מֵיטַב שָׂדֵהוּ וּמֵיטַב כַּרְמֹו יְשַׁלֵּם׃ ס 5 כִּי־תֵצֵא אֵשׁ וּמָצְאָה קֹצִים וְנֶאֱכַל גָּדִישׁ אֹו הַקָּמָה אֹו הַשָּׂדֶה שַׁלֵּם יְשַׁלֵּם הַמַּבְעִר אֶת־הַבְּעֵרָה׃ ס 6 כִּי־יִתֵּן אִישׁ אֶל־רֵעֵהוּ כֶּסֶף אֹו־כֵלִים לִשְׁמֹר וְגֻנַּב מִבֵּית הָאִישׁ אִמ־יִמָּצֵא הַגַּנָּב יְשַׁלֵּם שְׁנָיִם׃ 7 אִמ־לֹא יִמָּצֵא הַגַּנָּב וְנִקְרַב בַּעַל־הַבַּיִת אֶל־הָאֱלֹהִים אִמ־לֹא שָׁלַח יָדֹו בִּמְלֶאכֶת רֵעֵהוּ׃ 8 עַל־כָּל־דְּבַר־פֶּשַׁע עַל־שֹׁור עַל־חֲמֹור עַל־שֶׂה עַל־שַׂלְמָה עַל־כָּל־אֲבֵדָה אֲשֶׁר יֹאמַר כִּי־הוּא זֶה עַד הָאֱלֹהִים יָבֹא דְּבַר־שְׁנֵיהֶם אֲשֶׁר יַרְשִׁיעֻן אֱלֹהִים יְשַׁלֵּם שְׁנַיִם לְרֵעֵהוּ׃ ס 9 כִּי־יִתֵּן אִישׁ אֶל־רֵעֵהוּ חֲמֹור אֹו־שֹׁור אֹו־שֶׂה וְכָל־בְּהֵמָה לִשְׁמֹר וּמֵת אֹו־נִשְׁבַּר אֹו־נִשְׁבָּה אֵין רֹאֶה׃ 10 שְׁבֻעַת יְהוָה תִּהְיֶה בֵּין שְׁנֵיהֶם אִמ־לֹא שָׁלַח יָדֹו בִּמְלֶאכֶת רֵעֵהוּ וְלָקַח בְּעָלָיו וְלֹא יְשַׁלֵּם׃ 11 וְאִמ־גָּנֹב יִגָּנֵב מֵעִמֹּו יְשַׁלֵּם לִבְעָלָיו׃ 12 אִמ־טָרֹף יִטָּרֵף יְבִאֵהוּ עֵד הַטְּרֵפָה לֹא יְשַׁלֵּם׃ ף 13 וְכִי־יִשְׁאַל אִישׁ מֵעִם רֵעֵהוּ וְנִשְׁבַּר אֹו־מֵת בְּעָלָיו אֵינ־עִמֹּו שַׁלֵּם יְשַׁלֵּם׃ 14 אִמ־בְּעָלָיו עִמֹּו לֹא יְשַׁלֵּם אִמ־שָׂכִיר הוּא בָּא בִּשְׂכָרֹו׃ ס 15 וְכִי־יְפַתֶּה אִישׁ בְּתוּלָה אֲשֶׁר לֹא־אֹרָשָׂה וְשָׁכַב עִמָּהּ מָהֹר יִמְהָרֶנָּה לֹּו לְאִשָּׁה׃ 16 אִמ־מָאֵן יְמָאֵן אָבִיהָ לְתִתָּהּ לֹו כֶּסֶף יִשְׁקֹל כְּמֹהַר הַבְּתוּלֹת׃ ס 17 מְכַשֵּׁפָה לֹא תְחַיֶּה׃ ס 18 כָּל־שֹׁכֵב עִמ־בְּהֵמָה מֹות יוּמָת׃ ס 19 זֹבֵחַ לָאֱלֹהִים יָחֳרָם בִּלְתִּי לַיהוָה לְבַדֹּו׃ 20 וְגֵר לֹא־תֹונֶה וְלֹא תִלְחָצֶנּוּ כִּי־גֵרִים הֱיִיתֶם בְּאֶרֶץ מִצְרָיִם׃ 21 כָּל־אַלְמָנָה וְיָתֹום לֹא תְעַנּוּן׃ 22 אִמ־עַנֵּה תְעַנֶּה אֹתֹו כִּי אִמ־צָעֹק יִצְעַק אֵלַי שָׁמֹעַ אֶשְׁמַע צַעֲקָתֹו׃ 23 וְחָרָה אַפִּי וְהָרַגְתִּי אֶתְכֶם בֶּחָרֶב וְהָיוּ נְשֵׁיכֶם אַלְמָנֹות וּבְנֵיכֶם יְתֹמִים׃ ף 24 אִמ־כֶּסֶף ׀ תַּלְוֶה אֶת־עַמִּי אֶת־הֶעָנִי עִמָּךְ לֹא־תִהְיֶה לֹו כְּנֹשֶׁה לֹא־תְשִׂימוּן עָלָיו נֶשֶׁךְ׃ 25 אִמ־חָבֹל תַּחְבֹּל שַׂלְמַת רֵעֶךָ עַד־בֹּא הַשֶּׁמֶשׁ תְּשִׁיבֶנּוּ לֹו׃ 26 כִּי הִוא כְסוּתָה כְסוּתֹו לְבַדָּהּ הִוא שִׂמְלָתֹו לְעֹרֹו בַּמֶּה יִשְׁכָּב וְהָיָה כִּי־יִצְעַק אֵלַי וְשָׁמַעְתִּי כִּי־חַנּוּן אָנִי׃ ס 27 אֱלֹהִים לֹא תְקַלֵּל וְנָשִׂיא בְעַמְּךָ לֹא תָאֹר׃ 28 מְלֵאָתְךָ וְדִמְעֲךָ לֹא תְאַחֵר בְּכֹור בָּנֶיךָ תִּתֶּנ־לִּי׃ 29 כֵּנ־תַּעֲשֶׂה לְשֹׁרְךָ לְצֹאנֶךָ שִׁבְעַת יָמִים יִהְיֶה עִמ־אִמֹּו בַּיֹּום הַשְּׁמִינִי תִּתְּנֹו־לִי׃ 30 וְאַנְשֵׁי־קֹדֶשׁ
