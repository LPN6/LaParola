1 בְּרִית כָּרַתִּי לְעֵינָי וּמָה אֶתְבֹּונֵן עַל־בְּתוּלָה׃ 2 וּמֶה ׀ חֵלֶק אֱלֹוהַּ מִמָּעַל וְנַחֲלַת שַׁדַּי מִמְּרֹמִים׃ 3 הֲלֹא־אֵיד לְעַוָּל וְנֵכֶר לְפֹעֲלֵי אָוֶן׃ 4 הֲלֹא־הוּא יִרְאֶה דְרָכָי וְכָל־צְעָדַי יִסְפֹּור׃ 5 אִמ־הָלַכְתִּי עִמ־שָׁוְא וַתַּחַשׁ עַל־מִרְמָה רַגְלִי׃ 6 יִשְׁקְלֵנִי בְמֹאזְנֵי־צֶדֶק וְיֵדַע אֱלֹוהַּ תֻּמָּתִי׃ 7 אִם תִּטֶּה אַשֻּׁרִי מִנִּי הַדָּרֶךְ וְאַחַר עֵינַי הָלַךְ לִבִּי וּבְכַפַּי דָּבַק מֻאוּם׃ ף 8 אֶזְרְעָה וְאַחֵר יֹאכֵל וְצֶאֱצָאַי יְשֹׁרָשׁוּ׃ 9 אִמ־נִפְתָּה לִבִּי עַל־אִשָּׁה וְעַל־פֶּתַח רֵעִי אָרָבְתִּי׃ 10 תִּטְחַן לְאַחֵר אִשְׁתִּי וְעָלֶיהָ יִכְרְעוּן אֲחֵרִין׃ 11 כִּי־הוּא הִיא זִמָּה וְהִיא וְהוּא עָוֹן פְּלִילִים׃ 12 כִּי אֵשׁ הִיא עַד־אֲבַדֹּון תֹּאכֵל וּבְכָל־תְּבוּאָתִי תְשָׁרֵשׁ׃ 13 אִמ־אֶמְאַס מִשְׁפַּט עַבְדִּי וַאֲמָתִי בְּרִבָם עִמָּדִי׃ 14 וּמָה אֶעֱשֶׂה כִּי־יָקוּם אֵל וְכִי־יִפְקֹד מָה אֲשִׁיבֶנּוּ׃ 15 הֲלֹא־בַבֶּטֶן עֹשֵׂנִי עָשָׂהוּ וַיְכֻנֶנּוּ בָּרֶחֶם אֶחָד׃ 16 אִמ־אֶמְנַע מֵחֵפֶץ דַּלִּים וְעֵינֵי אַלְמָנָה אֲכַלֶּה׃ 17 וְאֹכַל פִּתִּי לְבַדִּי וְלֹא־אָכַל יָתֹום מִמֶּנָּה׃ 18 כִּי מִנְּעוּרַי גְּדֵלַנִי כְאָב וּמִבֶּטֶן אִמִּי אַנְחֶנָּה׃ 19 אִמ־אֶרְאֶה אֹובֵד מִבְּלִי לְבוּשׁ וְאֵין כְּסוּת לָאֶבְיֹון׃ 20 אִמ־לֹא בֵרֲכוּנִי חֲלָצֹו חֲלָצָיו וּמִגֵּז כְּבָשַׂי יִתְחַמָּם׃ 21 אִמ־הֲנִיפֹותִי עַל־יָתֹום יָדִי כִּי־אֶרְאֶה בַשַּׁעַר עֶזְרָתִי׃ 22 כְּתֵפִי מִשִּׁכְמָה תִפֹּול וְאֶזְרֹעִי מִקָּנָה תִשָּׁבֵר׃ 23 כִּי פַחַד אֵלַי אֵיד אֵל וּמִשְּׂאֵתֹו לֹא אוּכָל׃ 24 אִמ־שַׂמְתִּי זָהָב כִּסְלִי וְלַכֶּתֶם אָמַרְתִּי מִבְטַחִי׃ 25 אִמ־אֶשְׂמַח כִּי־רַב חֵילִי וְכִי־כַבִּיר מָצְאָה יָדִי׃ 26 אִמ־אֶרְאֶה אֹור כִּי יָהֵל וְיָרֵחַ יָקָר הֹלֵךְ׃ 27 וַיִּפְתְּ בַּסֵּתֶר לִבִּי וַתִּשַּׁק יָדִי לְפִי׃ 28 גַּמ־הוּא עָוֹן פְּלִילִי כִּי־כִחַשְׁתִּי לָאֵל מִמָּעַל׃ 29 אִמ־אֶשְׂמַח בְּפִיד מְשַׂנְאִי וְהִתְעֹרַרְתִּי כִּי־מְצָאֹו רָע׃ 30 וְלֹא־נָתַתִּי לַחֲטֹא חִכִּי לִשְׁאֹל בְּאָלָה נַפְשֹׁו׃ 31 אִמ־לֹא אָמְרוּ מְתֵי אָהֳלִי מִי־יִתֵּן מִבְּשָׂרֹו לֹא נִשְׂבָּע׃ 32 בַּחוּץ לֹא־יָלִין גֵּר דְּלָתַי לָאֹרַח אֶפְתָּח׃ 33 אִמ־כִּסִּיתִי כְאָדָם פְּשָׁעָי לִטְמֹון בְּחֻבִּי עֲוֹנִי׃ 34 כִּי אֶעֱרֹוץ ׀ הָמֹון רַבָּה וּבוּז־מִשְׁפָּחֹות יְחִתֵּנִי וָאֶדֹּם לֹא־אֵצֵא פָתַח׃ 35 מִי יִתֶּנ־לִי ׀ שֹׁמֵעַ לִי הֶנ־תָּוִי שַׁדַּי יַעֲנֵנִי וְסֵפֶר כָּתַב אִישׁ רִיבִי׃ 36 אִמ־לֹא עַל־שִׁכְמִי אֶשָּׂאֶנּוּ אֶעֶנְדֶנּוּ עֲטָרֹות לִי׃ 37 מִסְפַּר צְעָדַי אַגִּידֶנּוּ כְּמֹו־נָגִיד אֲקָרֲבֶנּוּ׃ 38 אִמ־עָלַי אַדְמָתִי תִזְעָק וְיַחַד תְּלָמֶיהָ יִבְכָּיוּן׃ 39 אִמ־כֹּחָהּ אָכַלְתִּי בְלִי־כָסֶף וְנֶפֶשׁ בְּעָלֶיהָ הִפָּחְתִּי׃ 40 תַּחַת חִטָּה ׀ יֵצֵא חֹוחַ וְתַחַת־שְׂעֹרָה
