1 וְנַעֲמָן שַׂר־צְבָא מֶלֶכְ־אֲרָם הָיָה אִישׁ גָּדֹול לִפְנֵי אֲדֹנָיו וּנְשֻׂא פָנִים כִּי־בֹו נָתַנ־יְהוָה תְּשׁוּעָה לַאֲרָם וְהָאִישׁ הָיָה גִּבֹּור חַיִל מְצֹרָע׃ 2 וַאֲרָם יָצְאוּ גְדוּדִים וַיִּשְׁבּוּ מֵאֶרֶץ יִשְׂרָאֵל נַעֲרָה קְטַנָּה וַתְּהִי לִפְנֵי אֵשֶׁת נַעֲמָן׃ 3 וַתֹּאמֶר אֶל־גְּבִרְתָּהּ אַחֲלֵי אֲדֹנִי לִפְנֵי הַנָּבִיא אֲשֶׁר בְּשֹׁמְרֹון אָז יֶאֱסֹף אֹתֹו מִצָּרַעְתֹּו׃ אֵלֶיךָ אֶת־נַעֲמָן עַבְדִּי וַאֲסַפְתֹּו מִצָּרַעְתֹּו׃ 7 וַיְהִי כִּקְרֹא מֶלֶכְ־יִשְׂרָאֵל אֶת־הַסֵּפֶר וַיִּקְרַע בְּגָדָיו וַיֹּאמֶר הַאֱלֹהִים אָנִי לְהָמִית וּלְהַחֲיֹות כִּי־זֶה שֹׁלֵחַ אֵלַי לֶאֱסֹף אִישׁ מִצָּרַעְתֹּו כִּי אַכְ־דְּעוּ־נָא וּרְאוּ כִּי־מִתְאַנֶּה הוּא לִי׃ 8 וַיְהִי כִּשְׁמֹעַ ׀ אֱלִישָׁע אִישׁ־הָאֱלֹהִים כִּי־קָרַע מֶלֶכְ־יִשְׂרָאֵל אֶת־בְּגָדָיו וַיִּשְׁלַח אֶל־הַמֶּלֶךְ לֵאמֹר לָמָּה קָרַעְתָּ בְּגָדֶיךָ יָבֹא־נָא אֵלַי וְיֵדַע כִּי יֵשׁ נָבִיא בְּיִשְׂרָאֵל׃ 9 וַיָּבֹא נַעֲמָן בְּסוּסֹו בְּסוּסָיו וּבְרִכְבֹּו וַיַּעֲמֹד פֶּתַח־הַבַּיִת לֶאֱלִישָׁע׃ אֵלֶיךָ רְחַץ וּטְהָר׃ 14 וַיֵּרֶד וַיִּטְבֹּל בַּיַּרְדֵּן שֶׁבַע פְּעָמִים כִּדְבַר אִישׁ הָאֱלֹהִים וַיָּשָׁב בְּשָׂרֹו כִּבְשַׂר נַעַר קָטֹן וַיִּטְהָר׃ 15 וַיָּשָׁב אֶל־אִישׁ הָאֱלֹהִים הוּא
