1 וַיִּסְעוּ בְּנֵי יִשְׂרָאֵל וַיַּחֲנוּ בְּעַרְבֹות מֹואָב מֵעֵבֶר לְיַרְדֵּן יְרֵחֹו׃ ס 2 וַיַּרְא בָּלָק בֶּנ־צִפֹּור אֵת כָּל־אֲשֶׁר־עָשָׂה יִשְׂרָאֵל לָאֱמֹרִי׃ 3 וַיָּגָר מֹואָב מִפְּנֵי הָעָם מְאֹד כִּי רַב־הוּא וַיָּקָץ מֹואָב מִפְּנֵי בְּנֵי יִשְׂרָאֵל׃ 4 וַיֹּאמֶר מֹואָב אֶל־זִקְנֵי מִדְיָן עַתָּה יְלַחֲכוּ הַקָּהָל אֶת־כָּל־סְבִיבֹתֵינוּ כִּלְחֹךְ הַשֹּׁור אֵת יֶרֶק הַשָּׂדֶה וּבָלָק בֶּנ־צִפֹּור מֶלֶךְ לְמֹואָב בָּעֵת הַהִוא׃ 5 וַיִּשְׁלַח מַלְאָכִים אֶל־בִּלְעָם בֶּנ־בְּעֹור פְּתֹורָה אֲשֶׁר עַל־הַנָּהָר אֶרֶץ בְּנֵי־עַמֹּו לִקְרֹא־לֹו לֵאמֹר הִנֵּה עַם יָצָא מִמִּצְרַיִם הִנֵּה כִסָּה אֶת־עֵין הָאָרֶץ וְהוּא יֹשֵׁב מִמֻּלִי׃ 6 וְעַתָּה לְכָה־נָּא אָרָה־לִּי אֶת־הָעָם הַזֶּה כִּי־עָצוּם הוּא מִמֶּנִּי אוּלַי אוּכַל נַכֶּה־בֹּו וַאֲגָרְשֶׁנּוּ מִנ־הָאָרֶץ כִּי יָדַעְתִּי אֵת אֲשֶׁר־תְּבָרֵךְ מְבֹרָךְ וַאֲשֶׁר תָּאֹר יוּאָר׃ 7 וַיֵּלְכוּ זִקְנֵי מֹואָב וְזִקְנֵי מִדְיָן וּקְסָמִים בְּיָדָם וַיָּבֹאוּ אֶל־בִּלְעָם וַיְדַבְּרוּ אֵלָיו דִּבְרֵי בָלָק׃ 8 וַיֹּאמֶר אֲלֵיהֶם לִינוּ פֹה הַלַּיְלָה וַהֲשִׁבֹתִי אֶתְכֶם דָּבָר כַּאֲשֶׁר יְדַבֵּר יְהוָה אֵלָי וַיֵּשְׁבוּ שָׂרֵי־מֹואָב עִמ־בִּלְעָם׃ 9 וַיָּבֹא אֱלֹהִים אֶל־בִּלְעָם וַיֹּאמֶר מִי הָאֲנָשִׁים הָאֵלֶּה עִמָּךְ׃ 10 וַיֹּאמֶר בִּלְעָם אֶל־הָאֱלֹהִים בָּלָק בֶּנ־צִפֹּר מֶלֶךְ מֹואָב שָׁלַח אֵלָי׃ 11 הִנֵּה הָעָם הַיֹּצֵא מִמִּצְרַיִם וַיְכַס אֶת־עֵין הָאָרֶץ עַתָּה לְכָה קָבָה־לִּי אֹתֹו אוּלַי אוּכַל לְהִלָּחֶם בֹּו וְגֵרַשְׁתִּיו׃ 12 וַיֹּאמֶר אֱלֹהִים אֶל־בִּלְעָם לֹא תֵלֵךְ עִמָּהֶם לֹא תָאֹר אֶת־הָעָם כִּי בָרוּךְ הוּא׃ 13 וַיָּקָם בִּלְעָם בַּבֹּקֶר וַיֹּאמֶר אֶל־שָׂרֵי בָלָק לְכוּ אֶל־אַרְצְכֶם כִּי מֵאֵן יְהוָה לְתִתִּי לַהֲלֹךְ עִמָּכֶם׃ 14 וַיָּקוּמוּ שָׂרֵי מֹואָב וַיָּבֹאוּ אֶל־בָּלָק וַיֹּאמְרוּ מֵאֵן בִּלְעָם הֲלֹךְ עִמָּנוּ׃ 15 וַיֹּסֶף עֹוד בָּלָק שְׁלֹחַ שָׂרִים רַבִּים וְנִכְבָּדִים מֵאֵלֶּה׃ 16 וַיָּבֹאוּ אֶל־בִּלְעָם וַיֹּאמְרוּ לֹו כֹּה אָמַר בָּלָק בֶּנ־צִפֹּור אַל־נָא תִמָּנַע מֵהֲלֹךְ אֵלָי׃ 17 כִּי־כַבֵּד אֲכַבֶּדְךָ מְאֹד וְכֹל אֲשֶׁר־תֹּאמַר אֵלַי אֶעֱשֶׂה וּלְכָה־נָּא קָבָה־לִּי אֵת הָעָם הַזֶּה׃ 18 וַיַּעַן בִּלְעָם וַיֹּאמֶר אֶל־עַבְדֵי בָלָק אִמ־יִתֶּנ־לִי בָלָק מְלֹא בֵיתֹו כֶּסֶף וְזָהָב לֹא אוּכַל לַעֲבֹר אֶת־פִּי יְהוָה אֱלֹהָי לַעֲשֹׂות קְטַנָּה אֹו גְדֹולָה׃ 19 וְעַתָּה שְׁבוּ נָא בָזֶה גַּמ־אַתֶּם הַלָּיְלָה וְאֵדְעָה מַה־יֹּסֵף יְהוָה דַּבֵּר עִמִּי׃ 20 וַיָּבֹא אֱלֹהִים ׀ אֶל־בִּלְעָם לַיְלָה וַיֹּאמֶר לֹו אִמ־לִקְרֹא לְךָ בָּאוּ הָאֲנָשִׁים קוּם לֵךְ אִתָּם וְאַךְ אֶת־הַדָּבָר אֲשֶׁר־אֲדַבֵּר אֵלֶיךָ אֹתֹו תַעֲשֶׂה׃ 21 וַיָּקָם בִּלְעָם בַּבֹּקֶר וַיַּחֲבֹשׁ אֶת־אֲתֹנֹו וַיֵּלֶךְ עִמ־שָׂרֵי מֹואָב׃ 22 וַיִּחַר־אַף אֱלֹהִים כִּי־הֹולֵךְ הוּא וַיִּתְיַצֵּב מַלְאַךְ יְהוָה בַּדֶּרֶךְ לְשָׂטָן לֹו וְהוּא רֹכֵב עַל־אֲתֹנֹו וּשְׁנֵי נְעָרָיו עִמֹּו׃ 23 וַתֵּרֶא הָאָתֹון אֶת־מַלְאַךְ יְהוָה נִצָּב בַּדֶּרֶךְ וְחַרְבֹּו שְׁלוּפָה בְּיָדֹו וַתֵּט הָאָתֹון מִנ־הַדֶּרֶךְ וַתֵּלֶךְ בַּשָּׂדֶה וַיַּךְ בִּלְעָם אֶת־הָאָתֹון לְהַטֹּתָהּ הַדָּרֶךְ׃ 24 וַיַּעֲמֹד מַלְאַךְ יְהוָה בְּמִשְׁעֹול הַכְּרָמִים גָּדֵר מִזֶּה וְגָדֵר מִזֶּה׃ 25 וַתֵּרֶא הָאָתֹון אֶת־מַלְאַךְ יְהוָה וַתִּלָּחֵץ אֶל־הַקִּיר וַתִּלְחַץ אֶת־רֶגֶל בִּלְעָם אֶל־הַקִּיר וַיֹּסֶף לְהַכֹּתָהּ׃ 26 וַיֹּוסֶף מַלְאַכְ־יְהוָה עֲבֹור וַיַּעֲמֹד בְּמָקֹום צָר אֲשֶׁר אֵינ־דֶּרֶךְ לִנְטֹות יָמִין וּשְׂמֹאול׃ 27 וַתֵּרֶא הָאָתֹון אֶת־מַלְאַךְ יְהוָה וַתִּרְבַּץ תַּחַת בִּלְעָם וַיִּחַר־אַף בִּלְעָם וַיַּךְ אֶת־הָאָתֹון בַּמַּקֵּל׃ 28 וַיִּפְתַּח יְהוָה אֶת־פִּי הָאָתֹון וַתֹּאמֶר לְבִלְעָם מֶה־עָשִׂיתִי לְךָ כִּי הִכִּיתַנִי זֶה שָׁלֹשׁ רְגָלִים׃ 29 וַיֹּאמֶר בִּלְעָם לָאָתֹון כִּי הִתְעַלַּלְתְּ בִּי לוּ יֶשׁ־חֶרֶב בְּיָדִי כִּי עַתָּה הֲרַגְתִּיךְ׃ 30 וַתֹּאמֶר הָאָתֹון אֶל־בִּלְעָם הֲלֹוא אָנֹכִי אֲתֹנְךָ אֲשֶׁר־רָכַבְתָּ עָלַי מֵעֹודְךָ עַד־הַיֹּום הַזֶּה הַהַסְכֵּן הִסְכַּנְתִּי לַעֲשֹׂות לְךָ כֹּה וַיֹּאמֶר לֹא׃ 31 וַיְגַל יְהוָה אֶת־עֵינֵי בִלְעָם וַיַּרְא אֶת־מַלְאַךְ יְהוָה נִצָּב בַּדֶּרֶךְ וְחַרְבֹּו שְׁלֻפָה בְּיָדֹו וַיִּקֹּד וַיִּשְׁתַּחוּ לְאַפָּיו׃ 32 וַיֹּאמֶר אֵלָיו מַלְאַךְ יְהוָה עַל־מָה הִכִּיתָ אֶת־אֲתֹנְךָ זֶה שָׁלֹושׁ רְגָלִים הִנֵּה אָנֹכִי יָצָאתִי לְשָׂטָן כִּי־יָרַט הַדֶּרֶךְ לְנֶגְדִּי׃ 33 וַתִּרְאַנִי הָאָתֹון וַתֵּט לְפָנַי זֶה שָׁלֹשׁ רְגָלִים אוּלַי נָטְתָה מִפָּנַי כִּי עַתָּה גַּמ־אֹתְכָה הָרַגְתִּי וְאֹותָהּ הֶחֱיֵיתִי׃ 34 וַיֹּאמֶר בִּלְעָם אֶל־מַלְאַךְ יְהוָה חָטָאתִי כִּי לֹא יָדַעְתִּי כִּי אַתָּה נִצָּב לִקְרָאתִי בַּדָּרֶךְ וְעַתָּה אִמ־רַע בְּעֵינֶיךָ אָשׁוּבָה לִּי׃ 35 וַיֹּאמֶר מַלְאַךְ יְהוָה אֶל־בִּלְעָם לֵךְ עִמ־הָאֲנָשִׁים וְאֶפֶס אֶת־הַדָּבָר אֲשֶׁר־אֲדַבֵּר אֵלֶיךָ אֹתֹו תְדַבֵּר וַיֵּלֶךְ בִּלְעָם עִמ־שָׂרֵי בָלָק׃ 36 וַיִּשְׁמַע בָּלָק כִּי בָא בִלְעָם וַיֵּצֵא לִקְרָאתֹו אֶל־עִיר מֹואָב אֲשֶׁר עַל־גְּבוּל אַרְנֹן אֲשֶׁר בִּקְצֵה הַגְּבוּל׃ 37 וַיֹּאמֶר בָּלָק אֶל־בִּלְעָם הֲלֹא שָׁלֹחַ שָׁלַחְתִּי אֵלֶיךָ לִקְרֹא־לָךְ לָמָּה לֹא־הָלַכְתָּ אֵלָי הַאֻמְנָם לֹא אוּכַל כַּבְּדֶךָ׃ 38 וַיֹּאמֶר בִּלְעָם אֶל־בָּלָק הִנֵּה־בָאתִי אֵלֶיךָ עַתָּה הֲיָכֹול אוּכַל דַּבֵּר מְאוּמָה הַדָּבָר אֲשֶׁר יָשִׂים אֱלֹהִים בְּפִי אֹתֹו אֲדַבֵּר׃ 39 וַיֵּלֶךְ בִּלְעָם עִמ־בָּלָק וַיָּבֹאוּ קִרְיַת חֻצֹות׃ 40 וַיִּזְבַּח בָּלָק בָּקָר וָצֹאן וַיְשַׁלַּח לְבִלְעָם וְלַשָּׂרִים אֲשֶׁר אִתֹּו׃ 41 וַיְהִי בַבֹּקֶר וַיִּקַּח בָּלָק אֶת־בִּלְעָם וַיַּעֲלֵהוּ בָּמֹות בָּעַל וַיַּרְא מִשָּׁם קְצֵה הָעָם׃
