9
1 וַיַּעַן אִיֹּוב וַיֹּאמַר׃ 2 אָמְנָם יָדַעְתִּי כִי־כֵן וּמַה־יִּצְדַּק אֱנֹושׁ עִמ־אֵל׃ 3 אִמ־יַחְפֹּץ לָרִיב עִמֹּו לֹא־יַעֲנֶנּוּ אַחַת מִנִּי־אָלֶף׃ 4 חֲכַם לֵבָב וְאַמִּיץ כֹּחַ מִי־הִקְשָׁה אֵלָיו וַיִּשְׁלָם׃ 5 הַמַּעְתִּיק הָרִים וְלֹא יָדָעוּ אֲשֶׁר הֲפָכָם בְּאַפֹּו׃ 6 הַמַּרְגִּיז אֶרֶץ מִמְּקֹומָהּ וְעַמּוּדֶיהָ יִתְפַלָּצוּן׃ 7 הָאֹמֵר לַחֶרֶס וְלֹא יִזְרָח וּבְעַד כֹּוכָבִים יַחְתֹּם׃ 8 נֹטֶה שָׁמַיִם לְבַדֹּו וְדֹורֵךְ עַל־בָּמֳתֵי יָם׃ 9 עֹשֶׂה־עָשׁ כְּסִיל וְכִימָה וְחַדְרֵי תֵמָן׃ 10 עֹשֶׂה גְדֹלֹות עַד־אֵין חֵקֶר וְנִפְלָאֹות עַד־אֵין מִסְפָּר׃ 11 הֵן יַעֲבֹר עָלַי וְלֹא אֶרְאֶה וְיַחֲלֹף וְלֹא־אָבִין לֹו׃ 12 הֵן יַחְתֹּף מִי יְשִׁיבֶנּוּ מִי־יֹאמַר אֵלָיו מַה־תַּעֲשֶׂה׃ 13 אֱלֹוהַּ לֹא־יָשִׁיב אַפֹּו תַּחַתֹו תַּחְתָּיו שָׁחֲחוּ עֹזְרֵי רָהַב׃ 14 אַף כִּי־אָנֹכִי אֶעֱנֶנּוּ אֶבְחֲרָה דְבָרַי עִמֹּו׃ 15 אֲשֶׁר אִמ־צָדַקְתִּי לֹא אֶעֱנֶה לִמְשֹׁפְטִי אֶתְחַנָּן׃ 16 אִמ־קָרָאתִי וַיַּעֲנֵנִי לֹא־אַאֲמִין כִּי־יַאֲזִין קֹולִי׃ 17 אֲשֶׁר־בִּשְׂעָרָה יְשׁוּפֵנִי וְהִרְבָּה פְצָעַי חִנָּם׃ 18 לֹא־יִתְּנֵנִי הָשֵׁב רוּחִי כִּי יַשְׂבִּעַנִי מַמְּרֹרִים׃ 19 אִמ־לְכֹחַ אַמִּיץ הִנֵּה וְאִמ־לְמִשְׁפָּט מִי יֹועִידֵנִי׃ 20 אִמ־אֶצְדָּק פִּי יַרְשִׁיעֵנִי תָּמ־אָנִי וַיַּעְקְשֵׁנִי׃ 21 תָּמ־אָנִי לֹא־אֵדַע נַפְשִׁי אֶמְאַס חַיָּי׃ 22 אַחַת הִיא עַל־כֵּן אָמַרְתִּי תָּם וְרָשָׁע הוּא מְכַלֶּה׃ 23 אִמ־שֹׁוט יָמִית פִּתְאֹם לְמַסַּת נְקִיִּם יִלְעָג׃ 24 אֶרֶץ ׀ נִתְּנָה בְיַד־רָשָׁע פְּנֵי־שֹׁפְטֶיהָ יְכַסֶּה אִמ־לֹא אֵפֹוא מִי־הוּא׃ 25 וְיָמַי קַלּוּ מִנִּי־רָץ בָּרְחוּ לֹא־רָאוּ טֹובָה׃ 26 חָלְפוּ עִמ־אֳנִיֹּות אֵבֶה כְּנֶשֶׁר יָטוּשׂ עֲלֵי־אֹכֶל׃ 27 אִמ־אָמְרִי אֶשְׁכְּחָה שִׂיחִי אֶעֶזְבָה פָנַי וְאַבְלִיגָה׃ 28 יָגֹרְתִּי כָל־עַצְּבֹתָי יָדַעְתִּי כִּי־לֹא תְנַקֵּנִי׃ 29 אָנֹכִי אֶרְשָׁע לָמָּה־זֶּה הֶבֶל אִיגָע׃ 30 אִמ־הִתְרָחַצְתִּי בְמֹו בְמֵי־שָׁלֶג וַהֲזִכֹּותִי בְּבֹר כַּפָּי׃ 31 אָז בַּשַּׁחַת תִּטְבְּלֵנִי וְתִעֲבוּנִי שַׂלְמֹותָי׃ 32 כִּי־לֹא־אִישׁ כָּמֹנִי אֶעֱנֶנּוּ נָבֹוא יַחְדָּו בַּמִּשְׁפָּט׃ 33 לֹא יֵשׁ־בֵּינֵינוּ מֹוכִיחַ יָשֵׁת יָדֹו עַל־שְׁנֵינוּ׃ 34 יָסֵר מֵעָלַי שִׁבְטֹו וְאֵמָתֹו אַל־תְּבַעֲתַנִּי׃ 35 אַדַבְּרָה וְלֹא אִירָאֶנּוּ כִּי לֹא־כֵן אָנֹכִי עִמָּדִי׃
