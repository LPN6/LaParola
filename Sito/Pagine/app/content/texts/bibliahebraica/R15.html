1 וּבְנֵי רְאוּבֵן בְּכֹור־יִשְׂרָאֵל כִּי הוּא הַבְּכֹור וּבְחַלְּלֹו יְצוּעֵי אָבִיו נִתְּנָה בְּכֹרָתֹו לִבְנֵי יֹוסֵף בֶּנ־יִשְׂרָאֵל וְלֹא לְהִתְיַחֵשׂ לַבְּכֹרָה׃ 2 כִּי יְהוּדָה גָּבַר בְּאֶחָיו וּלְנָגִיד מִמֶּנּוּ וְהַבְּכֹרָה לְיֹוסֵף׃ ס 3 בְּנֵי רְאוּבֵן בְּכֹור יִשְׂרָאֵל חֲנֹוךְ וּפַלּוּא חֶצְרֹון וְכַרְמִי׃ 4 בְּנֵי יֹואֵל שְׁמַעְיָה בְנֹו גֹּוג בְּנֹו שִׁמְעִי בְנֹו׃ 5 מִיכָה בְנֹו רְאָיָה בְנֹו בַּעַל בְּנֹו׃ 6 בְּאֵרָה בְנֹו אֲשֶׁר הֶגְלָה תִּלְּגַת פִּלְנְאֶסֶר מֶלֶךְ אַשֻּׁר הוּא נָשִׂיא לָראוּבֵנִי׃ 7 וְאֶחָיו לְמִשְׁפְּחֹתָיו בְּהִתְיַחֵשׂ לְתֹלְדֹותָם הָרֹאשׁ יְעִיאֵל וּזְכַרְיָהוּ׃ 8 וּבֶלַע בֶּנ־עָזָז בֶּנ־שֶׁמַע בֶּנ־יֹואֵל הוּא יֹושֵׁב בַּעֲרֹעֵר וְעַד־נְבֹו וּבַעַל מְעֹון׃ 9 וְלַמִּזְרָח יָשַׁב עַד־לְבֹוא מִדְבָּרָה לְמִנ־הַנָּהָר פְּרָת כִּי מִקְנֵיהֶם רָבוּ בְּאֶרֶץ גִּלְעָד׃ 10 וּבִימֵי שָׁאוּל עָשׂוּ מִלְחָמָה עִמ־הַהַגְרִאִים וַיִּפְּלוּ בְּיָדָם וַיֵּשְׁבוּ בְּאָהֳלֵיהֶם עַל־כָּל־פְּנֵי מִזְרָח לַגִּלְעָד׃ ף 11 וּבְנֵי־גָד לְנֶגְדָּם יָשְׁבוּ בְּאֶרֶץ הַבָּשָׁן עַד־סַלְכָה׃ 12 יֹואֵל הָרֹאשׁ וְשָׁפָם הַמִּשְׁנֶה וְיַעְנַי וְשָׁפָט בַּבָּשָׁן׃ 13 וַאֲחֵיהֶם לְבֵית אֲבֹותֵיהֶם מִיכָאֵל וּמְשֻׁלָּם וְשֶׁבַע וְיֹורַי וְיַעְכָּן וְזִיעַ וָעֵבֶר שִׁבְעָה׃ ס 14 אֵלֶּה ׀ בְּנֵי אֲבִיחַיִל בֶּנ־חוּרִי בֶּנ־יָרֹוחַ בֶּנ־גִּלְעָד בֶּנ־מִיכָאֵל בֶּנ־יְשִׁישַׁי בֶּנ־יַחְדֹּו בֶּנ־בּוּז׃ 15 אֲחִי בֶּנ־עַבְדִּיאֵל בֶּנ־גּוּנִי רֹאשׁ לְבֵית אֲבֹותָם׃ 16 וַיֵּשְׁבוּ בַּגִּלְעָד בַּבָּשָׁן וּבִבְנֹתֶיהָ וּבְכָל־מִגְרְשֵׁי שָׁרֹון עַל־תֹוצְאֹותָם׃ 17 כֻּלָּם הִתְיַחְשׂוּ בִּימֵי יֹותָם מֶלֶכְ־יְהוּדָה וּבִימֵי יָרָבְעָם מֶלֶכְ־יִשְׂרָאֵל׃ ף 18 בְּנֵי־רְאוּבֵן וְגָדִי וַחֲצִי שֵׁבֶט־מְנַשֶּׁה מִנ־בְּנֵי־חַיִל אֲנָשִׁים נֹשְׂאֵי מָגֵן וְחֶרֶב וְדֹרְכֵי קֶשֶׁת וּלְמוּדֵי מִלְחָמָה אַרְבָּעִים וְאַרְבָּעָה אֶלֶף וּשְׁבַע־מֵאֹות וְשִׁשִּׁים יֹצְאֵי צָבָא׃ 19 וַיַּעֲשׂוּ מִלְחָמָה עִמ־הַהַגְרִיאִים וִיטוּר וְנָפִישׁ וְנֹודָב׃ 20 וַיֵּעָזְרוּ עֲלֵיהֶם וַיִּנָּתְנוּ בְיָדָם הַהַגְרִיאִים וְכֹל שֶׁעִמָּהֶם כִּי מֵאָה אָלֶף׃ 22 כִּי־חֲלָלִים רַבִּים נָפָלוּ כִּי מֵהָאֱלֹהִים הַמִּלְחָמָה וַיֵּשְׁבוּ תַחְתֵּיהֶם עַד־הַגֹּלָה׃ ף 23 וּבְנֵי חֲצִי שֵׁבֶט מְנַשֶּׁה יָשְׁבוּ בָּאָרֶץ מִבָּשָׁן עַד־בַּעַל חֶרְמֹון
