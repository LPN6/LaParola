21
1 וַיְהִי רָעָב בִּימֵי דָוִד שָׁלֹשׁ שָׁנִים שָׁנָה אַחֲרֵי שָׁנָה וַיְבַקֵּשׁ דָּוִד אֶת־פְּנֵי יְהוָה ס וַיֹּאמֶר יְהוָה אֶל־שָׁאוּל וְאֶל־בֵּית הַדָּמִים עַל־אֲשֶׁר־הֵמִית אֶת־הַגִּבְעֹנִים׃ 2 וַיִּקְרָא הַמֶּלֶךְ לַגִּבְעֹנִים וַיֹּאמֶר אֲלֵיהֶם וְהַגִּבְעֹנִים לֹא מִבְּנֵי יִשְׂרָאֵל הֵמָּה כִּי אִמ־מִיֶּתֶר הָאֱמֹרִי וּבְנֵי יִשְׂרָאֵל נִשְׁבְּעוּ לָהֶם וַיְבַקֵּשׁ שָׁאוּל לְהַכֹּתָם בְּקַנֹּאתֹו לִבְנֵי־יִשְׂרָאֵל וִיהוּדָה׃ 3 וַיֹּאמֶר דָּוִד אֶל־הַגִּבְעֹנִים מָה אֶעֱשֶׂה לָכֶם וּבַמָּה אֲכַפֵּר וּבָרְכוּ אֶת־נַחֲלַת יְהוָה׃ 4 וַיֹּאמְרוּ לֹו הַגִּבְעֹנִים אֵינ־לִי לָנוּ כֶּסֶף וְזָהָב עִמ־שָׁאוּל וְעִמ־בֵּיתֹו וְאֵינ־לָנוּ אִישׁ לְהָמִית בְּיִשְׂרָאֵל וַיֹּאמֶר מָה־אַתֶּם אֹמְרִים אֶעֱשֶׂה לָכֶם׃ 5 וַיֹּאמְרוּ אֶל־הַמֶּלֶךְ הָאִישׁ אֲשֶׁר כִּלָּנוּ וַאֲשֶׁר דִּמָּה־לָנוּ נִשְׁמַדְנוּ מֵהִתְיַצֵּב בְּכָל־גְּבֻל יִשְׂרָאֵל׃ 6 יְנָתָן יֻתַּנ־לָנוּ שִׁבְעָה אֲנָשִׁים מִבָּנָיו וְהֹוקַעֲנוּם לַיהוָה בְּגִבְעַת שָׁאוּל בְּחִיר יְהוָה ס וַיֹּאמֶר הַמֶּלֶךְ אֲנִי אֶתֵּן׃ 7 וַיַּחְמֹל הַמֶּלֶךְ עַל־מְפִי־בֹשֶׁת בֶּנ־יְהֹונָתָן בֶּנ־שָׁאוּל עַל־שְׁבֻעַת יְהוָה אֲשֶׁר בֵּינֹתָם בֵּין דָּוִד וּבֵין יְהֹונָתָן בֶּנ־שָׁאוּל׃ 8 וַיִּקַּח הַמֶּלֶךְ אֶת־שְׁנֵי בְּנֵי רִצְפָּה בַת־אַיָּה אֲשֶׁר יָלְדָה לְשָׁאוּל אֶת־אַרְמֹנִי וְאֶת־מְפִבֹשֶׁת וְאֶת־חֲמֵשֶׁת בְּנֵי מִיכַל בַּת־שָׁאוּל אֲשֶׁר יָלְדָה לְעַדְרִיאֵל בֶּנ־בַּרְזִלַּי הַמְּחֹלָתִי׃ 9 וַיִּתְּנֵם בְּיַד הַגִּבְעֹנִים וַיֹּקִיעֻם בָּהָר לִפְנֵי יְהוָה וַיִּפְּלוּ שִׁבַעְתָּיִם שְׁבַעְתָּם יָחַד וְהֵם וְהֵמָּה הֻמְתוּ בִּימֵי קָצִיר בָּרִאשֹׁנִים תְחִלַּת בִּתְחִלַּת קְצִיר שְׂעֹרִים׃ 10 וַתִּקַּח רִצְפָּה בַת־אַיָּה אֶת־הַשַּׂק וַתַּטֵּהוּ לָהּ אֶל־הַצּוּר מִתְּחִלַּת קָצִיר עַד נִתַּכְ־מַיִם עֲלֵיהֶם מִנ־הַשָּׁמָיִם וְלֹא־נָתְנָה עֹוף הַשָּׁמַיִם לָנוּחַ עֲלֵיהֶם יֹומָם וְאֶת־חַיַּת הַשָּׂדֶה לָיְלָה׃ 11 וַיֻּגַּד לְדָוִד אֵת אֲשֶׁר־עָשְׂתָה רִצְפָּה בַת־אַיָּה פִּלֶגֶשׁ שָׁאוּל׃ 12 וַיֵּלֶךְ דָּוִד וַיִּקַּח אֶת־עַצְמֹות שָׁאוּל וְאֶת־עַצְמֹות יְהֹונָתָן בְּנֹו מֵאֵת בַּעֲלֵי יָבֵישׁ גִּלְעָד אֲשֶׁר גָּנְבוּ אֹתָם מֵרְחֹב בֵּית־שַׁן אֲשֶׁר תְּלוּם תְּלָאוּם שָׁם שָׁמָּה הַפְּלִשְׁתִּים פְּלִשְׁתִּים בְּיֹום הַכֹּות פְּלִשְׁתִּים אֶת־שָׁאוּל בַּגִּלְבֹּעַ׃ 13 וַיַּעַל מִשָּׁם אֶת־עַצְמֹות שָׁאוּל וְאֶת־עַצְמֹות יְהֹונָתָן בְּנֹו וַיַּאַסְפוּ אֶת־עַצְמֹות הַמּוּקָעִים׃ 14 וַיִּקְבְּרוּ אֶת־עַצְמֹות־שָׁאוּל וִיהֹונָתָנ־בְּנֹו בְּאֶרֶץ בִּנְיָמִן בְּצֵלָע בְּקֶבֶר קִישׁ אָבִיו וַיַּעֲשׂוּ כֹּל אֲשֶׁר־צִוָּה הַמֶּלֶךְ וַיֵּעָתֵר אֱלֹהִים לָאָרֶץ אַחֲרֵי־כֵן׃ ף 15 וַתְּהִי־עֹוד מִלְחָמָה לַפְּלִשְׁתִּים אֶת־יִשְׂרָאֵל וַיֵּרֶד דָּוִד וַעֲבָדָיו עִמֹּו וַיִּלָּחֲמוּ אֶת־פְּלִשְׁתִּים וַיָּעַף דָּוִד׃ 16 וְיִשְׁבֹּו וְיִשְׁבִּי בְּנֹב אֲשֶׁר ׀ בִּילִידֵי הָרָפָה וּמִשְׁקַל קֵינֹו שְׁלֹשׁ מֵאֹות מִשְׁקַל נְחֹשֶׁת וְהוּא חָגוּר חֲדָשָׁה וַיֹּאמֶר לְהַכֹּות אֶת־דָּוִד׃ 17 וַיַּעֲזָר־לֹו אֲבִישַׁי בֶּנ־צְרוּיָה וַיַּךְ אֶת־הַפְּלִשְׁתִּי וַיְמִיתֵהוּ אָז נִשְׁבְּעוּ אַנְשֵׁי־דָוִד לֹו לֵאמֹר לֹא־תֵצֵא עֹוד אִתָּנוּ לַמִּלְחָמָה וְלֹא תְכַבֶּה אֶת־נֵר יִשְׂרָאֵל׃ ף 18 וַיְהִי אַחֲרֵי־כֵן וַתְּהִי־עֹוד הַמִּלְחָמָה בְּגֹוב עִמ־פְּלִשְׁתִּים אָז הִכָּה סִבְּכַי הַחֻשָׁתִי אֶת־סַף אֲשֶׁר בִּילִדֵי הָרָפָה׃ ף 19 וַתְּהִי־עֹוד הַמִּלְחָמָה בְּגֹוב עִמ־פְּלִשְׁתִּים וַיַּךְ אֶלְחָנָן בֶּנ־יַעְרֵי אֹרְגִים בֵּית הַלַּחְמִי אֵת גָּלְיָת הַגִּתִּי וְעֵץ חֲנִיתֹו כִּמְנֹור אֹרְגִים׃ ס 20 וַתְּהִי־עֹוד מִלְחָמָה בְּגַת וַיְהִי ׀ אִישׁ מָדִין מָדֹון וְאֶצְבְּעֹת יָדָיו וְאֶצְבְּעֹת רַגְלָיו שֵׁשׁ וָשֵׁשׁ עֶשְׂרִים וְאַרְבַּע מִסְפָּר וְגַמ־הוּא יֻלַּד לְהָרָפָה׃ 21 וַיְחָרֵף אֶת־יִשְׂרָאֵל וַיַּכֵּהוּ יְהֹונָתָן בֶּנ־שִׁמְעִי שִׁמְעָה אֲחִי דָוִד׃ 22 אֶת־אַרְבַּעַת אֵלֶּה יֻלְּדוּ לְהָרָפָה בְּגַת וַיִּפְּלוּ בְיַד־דָּוִד וּבְיַד עֲבָדָיו׃ ף
