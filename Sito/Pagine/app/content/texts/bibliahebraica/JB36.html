36
1 וַיֹּסֶף אֱלִיהוּא וַיֹּאמַר׃ 2 כַּתַּר־לִי זְעֵיר וַאֲחַוֶּךָּ כִּי עֹוד לֶאֱלֹוהַּ מִלִּים׃ 3 אֶשָּׂא דֵעִי לְמֵרָחֹוק וְלְפֹעֲלִי אֶתֵּנ־צֶדֶק׃ 4 כִּי־אָמְנָם לֹא־שֶׁקֶר מִלָּי תְּמִים דֵּעֹות עִמָּךְ׃ 5 הֶנ־אֵל כַּבִּיר וְלֹא יִמְאָס כַּבִּיר כֹּחַ לֵב׃ 6 לֹא־יְחַיֶּה רָשָׁע וּמִשְׁפַּט עֲנִיִּים יִתֵּן׃ 7 לֹא־יִגְרַע מִצַּדִּיק עֵינָיו וְאֶת־מְלָכִים לַכִּסֵּא וַיֹּשִׁיבֵם לָנֶצַח וַיִּגְבָּהוּ׃ 8 וְאִמ־אֲסוּרִים בַּזִּקִּים יִלָּכְדוּן בְּחַבְלֵי־עֹנִי׃ 9 וַיַּגֵּד לָהֶם פָּעֳלָם וּפִשְׁעֵיהֶם כִּי יִתְגַּבָּרוּ׃ 10 וַיִּגֶל אָזְנָם לַמּוּסָר וַיֹּאמֶר כִּי־יְשֻׁבוּן מֵאָוֶן׃ 11 אִמ־יִשְׁמְעוּ וְיַעֲבֹדוּ יְכַלּוּ יְמֵיהֶם בַּטֹּוב וּשְׁנֵיהֶם בַּנְּעִימִים׃ 12 וְאִמ־לֹא יִשְׁמְעוּ בְּשֶׁלַח יַעֲבֹרוּ וְיִגְוְעוּ כִּבְלִי־דָעַת׃ 13 וְחַנְפֵי־לֵב יָשִׂימוּ אָף לֹא יְשַׁוְּעוּ כִּי אֲסָרָם׃ 14 תָּמֹת בַּנֹּעַר נַפְשָׁם וְחַיָּתָם בַּקְּדֵשִׁים׃ 15 יְחַלֵּץ עָנִי בְעָנְיֹו וְיִגֶל בַּלַּחַץ אָזְנָם׃ 16 וְאַף הֲסִיתְךָ ׀ מִפִּי־צָר רַחַב לֹא־מוּצָק תַּחְתֶּיהָ וְנַחַת שֻׁלְחָנְךָ מָלֵא דָשֶׁן׃ 17 וְדִינ־רָשָׁע מָלֵאתָ דִּין וּמִשְׁפָּט יִתְמֹכוּ׃ 18 כִּי־חֵמָה פֶּנ־יְסִיתְךָ בְסָפֶק וְרָב־כֹּפֶר אַל־יַטֶּךָּ׃ 19 הֲיַעֲרֹךְ שׁוּעֲךָ לֹא בְצָר וְכֹל מַאֲמַצֵּי־כֹחַ׃ 20 אַל־תִּשְׁאַף הַלָּיְלָה לַעֲלֹות עַמִּים תַּחְתָּם׃ 21 הִשָּׁמֶר אַל־תֵּפֶן אֶל־אָוֶן כִּי־עַל־זֶה בָּחַרְתָּ מֵעֹנִי׃ 22 הֶנ־אֵל יַשְׂגִּיב בְּכֹחֹו מִי כָמֹהוּ מֹורֶה׃ 23 מִי־פָקַד עָלָיו דַּרְכֹּו וּמִי־אָמַר פָּעַלְתָּ עַוְלָה׃ 24 זְכֹר כִּי־תַשְׂגִּיא פָעֳלֹו אֲשֶׁר שֹׁרְרוּ אֲנָשִׁים׃ 25 כָּל־אָדָם חָזוּ־בֹו אֱנֹושׁ יַבִּיט מֵרָחֹוק׃ 26 הֶנ־אֵל שַׂגִּיא וְלֹא נֵדָע מִסְפַּר שָׁנָיו וְלֹא־חֵקֶר׃ 27 כִּי יְגָרַע נִטְפֵי־מָיִם יָזֹקּוּ מָטָר לְאֵדֹו׃ 28 אֲשֶׁר־יִזְּלוּ שְׁחָקִים יִרְעֲפוּ עֲלֵי ׀ אָדָם רָב׃ 29 אַף אִמ־יָבִין מִפְרְשֵׂי־עָב תְּשֻׁאֹות סֻכָּתֹו׃ 30 הֵנ־פָּרַשׂ עָלָיו אֹורֹו וְשָׁרְשֵׁי הַיָּם כִּסָּה׃ 31 כִּי־בָם יָדִין עַמִּים יִתֶּנ־אֹכֶל לְמַכְבִּיר׃ 32 עַל־כַּפַּיִם כִּסָּה־אֹור וַיְצַו עָלֶיהָ בְמַפְגִּיעַ׃ 33 יַגִּיד עָלָיו רֵעֹו מִקְנֶה אַף עַל־עֹולֶה׃
