1 לִבְנֵי עַמֹּון כֹּה אָמַר יְהוָה הֲבָנִים אֵין לְיִשְׂרָאֵל אִמ־יֹורֵשׁ אֵין לֹו מַדּוּעַ יָרַשׁ מַלְכָּם אֶת־גָּד וְעַמֹּו בְּעָרָיו יָשָׁב׃ 2 לָכֵן הִנֵּה יָמִים בָּאִים נְאֻמ־יְהוָה וְהִשְׁמַעְתִּי אֶל־רַבַּת בְּנֵי־עַמֹּון תְּרוּעַת מִלְחָמָה וְהָיְתָה לְתֵל שְׁמָמָה וּבְנֹתֶיהָ בָּאֵשׁ תִּצַּתְנָה וְיָרַשׁ יִשְׂרָאֵל אֶת־יֹרְשָׁיו אָמַר יְהוָה׃ 3 הֵילִילִי חֶשְׁבֹּון כִּי שֻׁדְּדָה־עַי צְעַקְנָה בְּנֹות רַבָּה חֲגֹרְנָה שַׂקִּים סְפֹדְנָה וְהִתְשֹׁוטַטְנָה בַּגְּדֵרֹות כִּי מַלְכָּם בַּגֹּולָה יֵלֵךְ כֹּהֲנָיו וְשָׂרָיו יַחְדָּיו׃ 4 מַה־תִּתְהַלְלִי בָּעֲמָקִים זָב עִמְקֵךְ הַבַּת הַשֹּׁובֵבָה הַבֹּטְחָה בְּאֹצְרֹתֶיהָ מִי יָבֹוא אֵלָי׃ 5 הִנְנִי מֵבִיא עָלַיִךְ פַּחַד נְאֻמ־אֲדֹנָי יְהוִה צְבָאֹות מִכָּל־סְבִיבָיִךְ וְנִדַּחְתֶּם אִישׁ לְפָנָיו וְאֵין מְקַבֵּץ לַנֹּדֵד׃ 6 וְאַחֲרֵי־כֵן אָשִׁיב אֶת־שְׁבוּת בְּנֵי־עַמֹּון נְאֻמ־יְהוָה׃ ס 7 לֶאֱדֹום כֹּה אָמַר יְהוָה צְבָאֹות הַאֵין עֹוד חָכְמָה בְּתֵימָן אָבְדָה עֵצָה מִבָּנִים נִסְרְחָה חָכְמָתָם׃ 8 נֻסוּ הָפְנוּ הֶעְמִיקוּ לָשֶׁבֶת יֹשְׁבֵי דְּדָן כִּי אֵיד עֵשָׂו הֵבֵאתִי עָלָיו עֵת פְּקַדְתִּיו׃ 9 אִמ־בֹּצְרִים בָּאוּ לָךְ לֹא יַשְׁאִרוּ עֹולֵלֹות אִמ־גַּנָּבִים בַּלַּיְלָה הִשְׁחִיתוּ דַיָּם׃ 10 כִּי־אֲנִי חָשַׂפְתִּי אֶת־עֵשָׂו גִּלֵּיתִי אֶת־מִסְתָּרָיו וְנֶחְבָּה לֹא יוּכָל שֻׁדַּד זַרְעֹו וְאֶחָיו וּשְׁכֵנָיו וְאֵינֶנּוּ׃ 11 עָזְבָה יְתֹמֶיךָ אֲנִי אֲחַיֶּה וְאַלְמְנֹתֶיךָ עָלַי תִּבְטָחוּ׃ ס 12 כִּי־כֹה ׀ אָמַר יְהוָה הִנֵּה אֲשֶׁר־אֵין מִשְׁפָּטָם לִשְׁתֹּות הַכֹּוס שָׁתֹו יִשְׁתּוּ וְאַתָּה הוּא נָקֹה תִּנָּקֶה לֹא תִנָּקֶה כִּי שָׁתֹה תִּשְׁתֶּה׃ 13 כִּי בִי נִשְׁבַּעְתִּי נְאֻמ־יְהוָה כִּי־לְשַׁמָּה לְחֶרְפָּה לְחֹרֶב וְלִקְלָלָה תִּהְיֶה בָצְרָה וְכָל־עָרֶיהָ תִהְיֶינָה לְחָרְבֹות עֹולָם׃ 14 שְׁמוּעָה שָׁמַעְתִּי מֵאֵת יְהוָה וְצִיר בַּגֹּויִם שָׁלוּחַ הִתְקַבְּצוּ וּבֹאוּ עָלֶיהָ וְקוּמוּ לַמִּלְחָמָה׃ 15 כִּי־הִנֵּה קָטֹן נְתַתִּיךָ בַּגֹּויִם בָּזוּי בָּאָדָם׃ 16 תִּפְלַצְתְּךָ הִשִּׁיא אֹתָךְ זְדֹון לִבֶּךָ שֹׁכְנִי בְּחַגְוֵי הַסֶּלַע תֹּפְשִׂי מְרֹום גִּבְעָה כִּי־תַגְבִּיהַ כַּנֶּשֶׁר קִנֶּךָ מִשָּׁם אֹורִידְךָ נְאֻמ־יְהוָה׃ 17 וְהָיְתָה אֱדֹום לְשַׁמָּה כֹּל עֹבֵר עָלֶיהָ יִשֹּׁם וְיִשְׁרֹק עַל־כָּל־מַכֹּותֶהָ׃ 18 כְּמַהְפֵּכַת סְדֹם וַעֲמֹרָה וּשְׁכֵנֶיהָ אָמַר יְהוָה לֹא־יֵשֵׁב שָׁם אִישׁ וְלֹא־יָגוּר בָּהּ בֶּנ־אָדָם׃ 19 הִנֵּה כְּאַרְיֵה יַעֲלֶה מִגְּאֹון הַיַּרְדֵּן אֶל־נְוֵה אֵיתָן כִּי־אַרְגִּיעָה אֲרִיצֶנּוּ מֵעָלֶיהָ וּמִי בָחוּר אֵלֶיהָ אֶפְקֹד כִּי מִי כָמֹונִי וּמִי יֹעִידֶנִּי וּמִי־זֶה רֹעֶה אֲשֶׁר יַעֲמֹד לְפָנָי׃ ס 20 לָכֵן שִׁמְעוּ עֲצַת־יְהוָה אֲשֶׁר יָעַץ אֶל־אֱדֹום וּמַחְשְׁבֹותָיו אֲשֶׁר חָשַׁב אֶל־יֹשְׁבֵי תֵימָן אִמ־לֹא יִסְחָבוּם צְעִירֵי הַצֹּאן אִמ־לֹא יַשִּׁים עֲלֵיהֶם נְוֵהֶם׃ 21 מִקֹּול נִפְלָם רָעֲשָׁה הָאָרֶץ צְעָקָה בְּיַמ־סוּף נִשְׁמַע קֹולָהּ׃ 22 הִנֵּה כַנֶּשֶׁר יַעֲלֶה וְיִדְאֶה וְיִפְרֹשׂ כְּנָפָיו עַל־בָּצְרָה וְהָיָה לֵב גִּבֹּורֵי אֱדֹום בַּיֹּום הַהוּא כְּלֵב אִשָּׁה מְצֵרָה׃ ס 23 לְדַמֶּשֶׂק בֹּושָׁה חֲמָת וְאַרְפָּד כִּי־שְׁמֻעָה רָעָה שָׁמְעוּ נָמֹגוּ בַּיָּם דְּאָגָה הַשְׁקֵט לֹא יוּכָל׃ 24 רָפְתָה דַמֶּשֶׂק הִפְנְתָה לָנוּס וְרֶטֶט ׀ הֶחֱזִיקָה צָרָה וַחֲבָלִים אֲחָזַתָּה כַּיֹּולֵדָה׃ 25 אֵיךְ לֹא־עֻזְּבָה עִיר תְּהִלָּה תְּהִלָּת קִרְיַת מְשֹׂושִׂי׃ 26 לָכֵן יִפְּלוּ בַחוּרֶיהָ בִּרְחֹבֹתֶיהָ וְכָל־אַנְשֵׁי הַמִּלְחָמָה יִדַּמּוּ בַּיֹּום הַהוּא נְאֻם יְהוָה צְבָאֹות׃ 27 וְהִצַּתִּי אֵשׁ בְּחֹומַת דַּמָּשֶׂק וְאָכְלָה אַרְמְנֹות בֶּנ־הֲדָד׃ ס 28 לְקֵדָר ׀ וּלְמַמְלְכֹות חָצֹור אֲשֶׁר הִכָּה נְבוּכַדְרֶאצֹּור נְבוּכַדְרֶאצַּר מֶלֶכְ־בָּבֶל כֹּה אָמַר יְהוָה קוּמוּ עֲלוּ אֶל־קֵדָר וְשָׁדְדוּ אֶת־בְּנֵי־קֶדֶם׃ 29 אָהֳלֵיהֶם וְצֹאנָם יִקָּחוּ יְרִיעֹותֵיהֶם וְכָל־כְּלֵיהֶם וּגְמַלֵּיהֶם יִשְׂאוּ לָהֶם וְקָרְאוּ עֲלֵיהֶם מָגֹור מִסָּבִיב׃ 30 נֻסוּ נֻּדוּ מְאֹד הֶעְמִיקוּ לָשֶׁבֶת יֹשְׁבֵי חָצֹור נְאֻמ־יְהוָה כִּי־יָעַץ עֲלֵיכֶם נְבוּכַדְרֶאצַּר מֶלֶכְ־בָּבֶל עֵצָה וְחָשַׁב עֲלֵיהֶם עֲלֵיכֶם מַחֲשָׁבָה׃ 31 קוּמוּ עֲלוּ אֶל־גֹּוי שְׁלֵיו יֹושֵׁב לָבֶטַח נְאֻמ־יְהוָה לֹא־דְלָתַיִם וְלֹא־בְרִיחַ לֹו בָּדָד יִשְׁכֹּנוּ׃ 32 וְהָיוּ גְמַלֵּיהֶם לָבַז וַהֲמֹון מִקְנֵיהֶם לְשָׁלָל וְזֵרִתִים לְכָל־רוּחַ קְצוּצֵי פֵאָה וּמִכָּל־עֲבָרָיו אָבִיא אֶת־אֵידָם נְאֻמ־יְהוָה׃ 33 וְהָיְתָה חָצֹור לִמְעֹון תַּנִּים שְׁמָמָה עַד־עֹולָם לֹא־יֵשֵׁב שָׁם אִישׁ וְלֹא־יָגוּר בָּהּ בֶּנ־אָדָם׃ ס 34 אֲשֶׁר הָיָה דְבַר־יְהוָה אֶל־יִרְמְיָהוּ הַנָּבִיא אֶל־עֵילָם בְּרֵאשִׁית מַלְכוּת צִדְקִיָּה מֶלֶכְ־יְהוּדָה לֵאמֹר׃ 35 כֹּה אָמַר יְהוָה צְבָאֹות הִנְנִי שֹׁבֵר אֶת־קֶשֶׁת עֵילָם רֵאשִׁית גְּבוּרָתָם׃ 36 וְהֵבֵאתִי אֶל־עֵילָם אַרְבַּע רוּחֹות מֵאַרְבַּע קְצֹות הַשָּׁמַיִם וְזֵרִתִים לְכֹל הָרֻחֹות הָאֵלֶּה וְלֹא־יִהְיֶה הַגֹּוי אֲשֶׁר לֹא־יָבֹוא שָׁם נִדְּחֵי עֹולָם עֵילָם׃ 37 וְהַחְתַּתִּי אֶת־עֵילָם לפְנֵי אֹיְבֵיהֶם וְלִפְנֵי ׀ מְבַקְשֵׁי נַפְשָׁם וְהֵבֵאתִי עֲלֵיהֶם ׀ רָעָה אֶת־חֲרֹון אַפִּי נְאֻמ־יְהוָה וְשִׁלַּחְתִּי אַחֲרֵיהֶם אֶת־הַחֶרֶב עַד כַּלֹּותִי אֹותָם׃ 38 וְשַׂמְתִּי כִסְאִי בְּעֵילָם וְהַאֲבַדְתִּי מִשָּׁם מֶלֶךְ וְשָׂרִים נְאֻמ־יְהוָה׃ 39 וְהָיָה ׀ בְּאַחֲרִית הַיָּמִים אָשׁוּב אָשִׁיב
