1 וֶאֱלִישָׁע הַנָּבִיא קָרָא לְאַחַד מִבְּנֵי הַנְּבִיאִים וַיֹּאמֶר לֹו חֲגֹר מָתְנֶיךָ וְקַח פַּךְ הַשֶּׁמֶן הַזֶּה בְּיָדֶךָ וְלֵךְ רָמֹת גִּלְעָד׃ 2 וּבָאתָ שָׁמָּה וּרְאֵה־שָׁם יֵהוּא בֶנ־יְהֹושָׁפָט בֶּנ־נִמְשִׁי וּבָאתָ וַהֲקֵמֹתֹו מִתֹּוך אֶחָיו וְהֵבֵיאתָ אֹתֹו חֶדֶר בְּחָדֶר׃ 3 וְלָקַחְתָּ פַכְ־הַשֶּׁמֶן וְיָצַקְתָּ עַל־רֹאשֹׁו וְאָמַרְתָּ כֹּה־אָמַר יְהוָה מְשַׁחְתִּיךָ לְמֶלֶךְ אֶל־יִשְׂרָאֵל וּפָתַחְתָּ הַדֶּלֶת וְנַסְתָּה וְלֹא תְחַכֶּה׃ 4 וַיֵּלֶךְ הַנַּעַר הַנַּעַר הַנָּבִיא רָמֹת גִּלְעָד׃ 5 וַיָּבֹא וְהִנֵּה שָׂרֵי הַחַיִל יֹשְׁבִים וַיֹּאמֶר דָּבָר לִי אֵלֶיךָ הַשָּׂר וַיֹּאמֶר יֵהוּא אֶל־מִי מִכֻּלָּנוּ וַיֹּאמֶר אֵלֶיךָ הַשָּׂר׃ 6 וַיָּקָם וַיָּבֹא הַבַּיְתָה וַיִּצֹק הַשֶּׁמֶן אֶל־רֹאשֹׁו וַיֹּאמֶר לֹו כֹּה־אָמַר יְהוָה אֱלֹהֵי יִשְׂרָאֵל מְשַׁחְתִּיךָ לְמֶלֶךְ אֶל־עַם יְהוָה אֶל־יִשְׂרָאֵל׃ 7 וְהִכִּיתָה אֶת־בֵּית אַחְאָב אֲדֹנֶיךָ וְנִקַּמְתִּי דְּמֵי ׀ עֲבָדַי הַנְּבִיאִים וּדְמֵי כָּל־עַבְדֵי יְהוָה מִיַּד אִיזָבֶל׃ 8 וְאָבַד 10 וְאֶת־אִיזֶבֶל יֹאכְלוּ הַכְּלָבִים בְּחֵלֶק יִזְרְעֶאל וְאֵין קֹבֵר וַיִּפְתַּח הַדֶּלֶת וַיָּנֹס׃ 11 וְיֵהוּא יָצָא אֶל־עַבְדֵי אֲדֹנָיו וַיֹּאמֶר לֹו הֲשָׁלֹום מַדּוּעַ בָּא־הַמְשֻׁגָּע הַזֶּה אֵלֶיךָ וַיֹּאמֶר אֲלֵיהֶם אַתֶּם יְדַעְתֶּם אֶת־הָאִישׁ וְאֶת־שִׂיחֹו׃ 12 וַיֹּאמְרוּ שֶׁקֶר הַגֶּד־נָא לָנוּ וַיֹּאמֶר כָּזֹאת וְכָזֹאת אָמַר אֵלַי לֵאמֹר כֹּה אָמַר יְהוָה מְשַׁחְתִּיךָ לְמֶלֶךְ
