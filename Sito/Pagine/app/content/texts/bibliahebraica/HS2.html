1 וְהָיָה מִסְפַּר בְּנֵי־יִשְׂרָאֵל כְּחֹול הַיָּם אֲשֶׁר לֹא־יִמַּד וְלֹא יִסָּפֵר וְהָיָה בִּמְקֹום אֲשֶׁר־יֵאָמֵר לָהֶם לֹא־עַמִּי אַתֶּם יֵאָמֵר לָהֶם בְּנֵי אֵל־חָי׃ 2 וְנִקְבְּצוּ בְּנֵי־יְהוּדָה וּבְנֵי־יִשְׂרָאֵל יַחְדָּו וְשָׂמוּ לָהֶם רֹאשׁ אֶחָד וְעָלוּ מִנ־הָאָרֶץ כִּי גָדֹול יֹום יִזְרְעֶאל׃ 3 אִמְרוּ לַאֲחֵיכֶם עַמִּי וְלַאֲחֹותֵיכֶם רֻחָמָה׃ 4 רִיבוּ בְאִמְּכֶם רִיבוּ כִּי־הִיא לֹא אִשְׁתִּי וְאָנֹכִי לֹא אִישָׁהּ וְתָסֵר זְנוּנֶיהָ מִפָּנֶיה וְנַאֲפוּפֶיהָ מִבֵּין שָׁדֶיהָ׃ 5 פֶּנ־אַפְשִׁיטֶנָּה עֲרֻמָּה וְהִצַּגְתִּיהָ כְּיֹום הִוָּלְדָהּ וְשַׂמְתִּיהָ כַמִּדְבָּר וְשַׁתִּהָ כְּאֶרֶץ צִיָּה וַהֲמִתִּיהָ בַּצָּמָא׃ 6 וְאֶת־בָּנֶיהָ לֹא אֲרַחֵם כִּי־בְנֵי זְנוּנִים הֵמָּה׃ 7 כִּי זָנְתָה אִמָּם הֹבִישָׁה הֹורָתָם כִּי אָמְרָה אֵלְכָה אַחֲרֵי מְאַהֲבַי נֹתְנֵי לַחְמִי וּמֵימַי צַמְרִי וּפִשְׁתִּי שַׁמְנִי וְשִׁקּוּיָי׃ 8 לָכֵן הִנְנִי־שָׂךְ אֶת־דַּרְכֵּךְ בַּסִּירִים וְגָדַרְתִּי אֶת־גְּדֵרָהּ וּנְתִיבֹותֶיהָ לֹא תִמְצָא׃ 9 וְרִדְּפָה אֶת־מְאַהֲבֶיהָ וְלֹא־תַשִּׂיג אֹתָם וּבִקְשָׁתַם וְלֹא תִמְצָא וְאָמְרָה אֵלְכָה וְאָשׁוּבָה אֶל־אִישִׁי הָרִאשֹׁון כִּי טֹוב לִי אָז מֵעָתָּה׃ 10 וְהִיא לֹא יָדְעָה כִּי אָנֹכִי נָתַתִּי לָהּ הַדָּגָן וְהַתִּירֹושׁ וְהַיִּצְהָר וְכֶסֶף הִרְבֵּיתִי לָהּ וְזָהָב עָשׂוּ לַבָּעַל׃ 11 לָכֵן אָשׁוּב וְלָקַחְתִּי דְגָנִי בְּעִתֹּו וְתִירֹושִׁי בְּמֹועֲדֹו וְהִצַּלְתִּי צַמְרִי וּפִשְׁתִּי לְכַסֹּות אֶת־עֶרְוָתָהּ׃ 12 וְעַתָּה אֲגַלֶּה אֶת־נַבְלֻתָהּ לְעֵינֵי מְאַהֲבֶיהָ וְאִישׁ לֹא־יַצִּילֶנָּה מִיָּדִי׃ 13 וְהִשְׁבַּתִּי כָּל־מְשֹׂושָׂהּ חַגָּהּ חָדְשָׁהּ וְשַׁבַּתָּהּ וְכֹל מֹועֲדָהּ׃ 14 וַהֲשִׁמֹּתִי גַּפְנָהּ וּתְאֵנָתָהּ אֲשֶׁר אָמְרָה אֶתְנָה הֵמָּה לִי אֲשֶׁר נָתְנוּ־לִי מְאַהֲבָי וְשַׂמְתִּים לְיַעַר וַאֲכָלָתַם חַיַּת הַשָּׂדֶה׃ 15 וּפָקַדְתִּי עָלֶיהָ אֶת־יְמֵי הַבְּעָלִים אֲשֶׁר תַּקְטִיר לָהֶם וַתַּעַד נִזְמָהּ וְחֶלְיָתָהּ וַתֵּלֶךְ אַחֲרֵי מְאַהֲבֶיהָ וְאֹתִי שָׁכְחָה נְאֻמ־יְהוָה׃ ף 16 לָכֵן הִנֵּה אָנֹכִי מְפַתֶּיהָ וְהֹלַכְתִּיהָ הַמִּדְבָּר וְדִבַּרְתִּי עַל־לִבָּהּ׃ 17 וְנָתַתִּי לָהּ אֶת־כְּרָמֶיהָ מִשָּׁם וְאֶת־עֵמֶק עָכֹור לְפֶתַח תִּקְוָה וְעָנְתָה שָּׁמָּה כִּימֵי נְעוּרֶיהָ וִּכְיֹום עֲלֹתָהּ מֵאֶרֶצ־מִצְרָיִם׃ ס 18 וְהָיָה בַיֹּומ־הַהוּא נְאֻמ־יְהוָה תִּקְרְאִי אִישִׁי וְלֹא־תִקְרְאִי־לִי עֹוד בַּעְלִי׃ 19 וַהֲסִרֹתִי אֶת־שְׁמֹות הַבְּעָלִים מִפִּיהָ וְלֹא־יִזָּכְרוּ עֹוד בִּשְׁמָם׃ 20 וְכָרַתִּי לָהֶם בְּרִית בַּיֹּום הַהוּא עִמ־חַיַּת הַשָּׂדֶה וְעִמ־עֹוף הַשָּׁמַיִם וְרֶמֶשׂ הָאֲדָמָה וְקֶשֶׁת וְחֶרֶב וּמִלְחָמָה אֶשְׁבֹּור מִנ־הָאָרֶץ וְהִשְׁכַּבְתִּים לָבֶטַח׃ 21 וְאֵרַשְׂתִּיךְ לִי לְעֹולָם וְאֵרַשְׂתִּיךְ לִי בְּצֶדֶק וּבְמִשְׁפָּט וּבְחֶסֶד וּבְרַחֲמִים׃ 22 וְאֵרַשְׂתִּיךְ לִי בֶּאֱמוּנָה וְיָדַעַתְּ אֶת־יְהוָה׃ ס 23 וְהָיָה ׀ בַּיֹּום הַהוּא אֶעֱנֶה נְאֻמ־יְהוָה אֶעֱנֶה אֶת־הַשָּׁמָיִם וְהֵם יַעֲנוּ אֶת־הָאָרֶץ׃ 24 וְהָאָרֶץ תַּעֲנֶה אֶת־הַדָּגָן וְאֶת־הַתִּירֹושׁ וְאֶת־הַיִּצְהָר וְהֵם יַעֲנוּ אֶת־יִזְרְעֶאל׃ 25 וּזְרַעְתִּיהָ לִּי בָּאָרֶץ וְרִחַמְתִּי אֶת־לֹא רֻחָמָה וְאָמַרְתִּי לְלֹא־עַמִּי
