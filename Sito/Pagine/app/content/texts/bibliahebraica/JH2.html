1 וַיְמַן יְהוָה דָּג גָּדֹול לִבְלֹעַ אֶת־יֹונָה וַיְהִי יֹונָה בִּמְעֵי הַדָּג שְׁלֹשָׁה יָמִים וּשְׁלֹשָׁה לֵילֹות׃ 2 וַיִּתְפַּלֵּל יֹונָה אֶל־יְהוָה אֱלֹהָיו מִמְּעֵי הַדָּגָה׃ 3 וַיֹּאמֶר קָרָאתִי מִצָּרָה לִי אֶל־יְהוָה וַיַּעֲנֵנִי מִבֶּטֶן שְׁאֹול שִׁוַּעְתִּי שָׁמַעְתָּ קֹולִי׃ 4 וַתַּשְׁלִיכֵנִי מְצוּלָה בִּלְבַב יַמִּים וְנָהָר יְסֹבְבֵנִי כָּל־מִשְׁבָּרֶיךָ וְגַלֶּיךָ עָלַי עָבָרוּ׃ 5 וַאֲנִי אָמַרְתִּי נִגְרַשְׁתִּי מִנֶּגֶד עֵינֶיךָ אַךְ אֹוסִיף לְהַבִּיט אֶל־הֵיכַל קָדְשֶׁךָ׃ 6 אֲפָפוּנִי מַיִם עַד־נֶפֶשׁ תְּהֹום יְסֹבְבֵנִי סוּף חָבוּשׁ לְרֹאשִׁי׃ 7 לְקִצְבֵי הָרִים יָרַדְתִּי הָאָרֶץ בְּרִחֶיהָ בַעֲדִי לְעֹולָם וַתַּעַל מִשַּׁחַת חַיַּי יְהוָה אֱלֹהָי׃ 8 בְּהִתְעַטֵּף עָלַי נַפְשִׁי אֶת־יְהוָה זָכָרְתִּי וַתָּבֹוא אֵלֶיךָ תְּפִלָּתִי אֶל־הֵיכַל קָדְשֶׁךָ׃ 9 מְשַׁמְּרִים הַבְלֵי־שָׁוְא חַסְדָּם יַעֲזֹבוּ׃ 10 וַאֲנִי בְּקֹול תֹּודָה אֶזְבְּחָה־לָּךְ אֲשֶׁר נָדַרְתִּי אֲשַׁלֵּמָה יְשׁוּעָתָה לַיהוָה׃ ס 11 וַיֹּאמֶר יְהוָה לַדָּג וַיָּקֵא אֶת־יֹונָה אֶל־הַיַּבָּשָׁה׃ ף
