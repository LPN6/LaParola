1 אָמַרְתִּי אֲנִי בְּלִבִּי לְכָה־נָּא אֲנַסְּכָה בְשִׂמְחָה וּרְאֵה בְטֹוב וְהִנֵּה גַמ־הוּא הָבֶל׃ 2 לִשְׂחֹוק אָמַרְתִּי מְהֹולָל וּלְשִׂמְחָה מַה־זֹּה עֹשָׂה׃ 3 תַּרְתִּי בְלִבִּי לִמְשֹׁוךְ בַּיַּיִן אֶת־בְּשָׂרִי וְלִבִּי נֹהֵג בַּחָכְמָה וְלֶאֱחֹז בְּסִכְלוּת עַד אֲשֶׁר־אֶרְאֶה אֵי־זֶה טֹוב לִבְנֵי הָאָדָם אֲשֶׁר יַעֲשׂוּ תַּחַת הַשָּׁמַיִם מִסְפַּר יְמֵי חַיֵּיהֶם׃ 4 הִגְדַּלְתִּי מַעֲשָׂי בָּנִיתִי לִי בָּתִּים נָטַעְתִּי לִי כְּרָמִים׃ 5 עָשִׂיתִי לִי גַּנֹּות וּפַרְדֵּסִים וְנָטַעְתִּי בָהֶם עֵץ כָּל־פֶּרִי׃ 6 עָשִׂיתִי לִי בְּרֵכֹות מָיִם לְהַשְׁקֹות מֵהֶם יַעַר צֹומֵחַ עֵצִים׃ 7 קָנִיתִי עֲבָדִים וּשְׁפָחֹות וּבְנֵי־בַיִת הָיָה לִי גַּם מִקְנֶה בָקָר וָצֹאן הַרְבֵּה הָיָה לִי מִכֹּל שֶׁהָיוּ לְפָנַי בִּירוּשָׁלִָם׃ 8 כָּנַסְתִּי לִי גַּמ־כֶּסֶף וְזָהָב וּסְגֻלַּת מְלָכִים וְהַמְּדִינֹות עָשִׂיתִי לִי שָׁרִים וְשָׁרֹות וְתַעֲנוּגֹת בְּנֵי הָאָדָם שִׁדָּה וְשִׁדֹּות׃ 9 וְגָדַלְתִּי וְהֹוסַפְתִּי מִכֹּל שֶׁהָיָה לְפָנַי בִּירוּשָׁלִָם אַף חָכְמָתִי עָמְדָה לִּי׃ 10 וְכֹל אֲשֶׁר שָׁאֲלוּ עֵינַי לֹא אָצַלְתִּי מֵהֶם לֹא־מָנַעְתִּי אֶת־לִבִּי מִכָּל־שִׂמְחָה כִּי־לִבִּי שָׂמֵחַ מִכָּל־עֲמָלִי וְזֶה־הָיָה חֶלְקִי מִכָּל־עֲמָלִי׃ 11 וּפָנִיתִי אֲנִי בְּכָל־מַעֲשַׂי שֶׁעָשׂוּ יָדַי וּבֶעָמָל שֶׁעָמַלְתִּי לַעֲשֹׂות וְהִנֵּה הַכֹּל הֶבֶל וּרְעוּת רוּחַ וְאֵין יִתְרֹון תַּחַת הַשָּׁמֶשׁ׃ 12 וּפָנִיתִי אֲנִי לִרְאֹות חָכְמָה וְהֹולֵלֹות וְסִכְלוּת כִּי ׀ מֶה הָאָדָם שֶׁיָּבֹוא אַחֲרֵי הַמֶּלֶךְ אֵת אֲשֶׁר־כְּבָר עָשׂוּהוּ׃ 13 וְרָאִיתִי אָנִי שֶׁיֵּשׁ יִתְרֹון לַחָכְמָה מִנ־הַסִּכְלוּת כִּיתְרֹון הָאֹור מִנ־הַחֹשֶׁךְ׃ 14 הֶחָכָם עֵינָיו בְּרֹאשֹׁו וְהַכְּסִיל בַּחֹשֶׁךְ הֹולֵךְ וְיָדַעְתִּי גַמ־אָנִי שֶׁמִּקְרֶה אֶחָד יִקְרֶה אֶת־כֻּלָּם׃ 15 וְאָמַרְתִּי אֲנִי בְּלִבִּי כְּמִקְרֵה הַכְּסִיל גַּמ־אֲנִי יִקְרֵנִי וְלָמָּה חָכַמְתִּי אֲנִי אָז יֹותֵר וְדִבַּרְתִּי בְלִבִּי שֶׁגַּמ־זֶה הָבֶל׃ 16 כִּי אֵין זִכְרֹון לֶחָכָם עִמ־הַכְּסִיל לְעֹולָם בְּשֶׁכְּבָר הַיָּמִים הַבָּאִים הַכֹּל נִשְׁכָּח וְאֵיךְ יָמוּת הֶחָכָם עִמ־הַכְּסִיל׃ 17 וְשָׂנֵאתִי אֶת־הַחַיִּים כִּי רַע עָלַי הַמַּעֲשֶׂה שֶׁנַּעֲשָׂה תַּחַת הַשָּׁמֶשׁ כִּי־הַכֹּל הֶבֶל וּרְעוּת רוּחַ׃ 18 וְשָׂנֵאתִי אֲנִי אֶת־כָּל־עֲמָלִי שֶׁאֲנִי עָמֵל תַּחַת הַשָּׁמֶשׁ שֶׁאַנִּיחֶנּוּ לָאָדָם שֶׁיִּהְיֶה אַחֲרָי׃ 19 וּמִי יֹודֵעַ הֶחָכָם יִהְיֶה אֹו סָכָל וְיִשְׁלַט בְּכָל־עֲמָלִי שֶׁעָמַלְתִּי וְשֶׁחָכַמְתִּי תַּחַת הַשָּׁמֶשׁ גַּמ־זֶה הָבֶל׃ 20 וְסַבֹּותִי אֲנִי לְיַאֵשׁ אֶת־לִבִּי עַל כָּל־הֶעָמָל שֶׁעָמַלְתִּי תַּחַת הַשָּׁמֶשׁ׃ 21 כִּי־יֵשׁ אָדָם שֶׁעֲמָלֹו בְּחָכְמָה וּבְדַעַת וּבְכִשְׁרֹון וּלְאָדָם שֶׁלֹּא עָמַל־בֹּו יִתְּנֶנּוּ חֶלְקֹו גַּמ־זֶה הֶבֶל וְרָעָה רַבָּה׃ 22 כִּי מֶה־הֹוֶה לָאָדָם בְּכָל־עֲמָלֹו וּבְרַעְיֹון לִבֹּו שֶׁהוּא עָמֵל תַּחַת הַשָּׁמֶשׁ׃ 23 כִּי כָל־יָמָיו מַכְאֹבִים וָכַעַס עִנְיָנֹו גַּמ־בַּלַּיְלָה לֹא־שָׁכַב לִבֹּו גַּמ־זֶה הֶבֶל הוּא׃ 24 אֵינ־טֹוב בָּאָדָם שֶׁיֹּאכַל וְשָׁתָה וְהֶרְאָה אֶת־נַפְשֹׁו טֹוב בַּעֲמָלֹו גַּמ־זֹה רָאִיתִי אָנִי כִּי מִיַּד הָאֱלֹהִים הִיא׃ 25 כִּי מִי יֹאכַל וּמִי יָחוּשׁ חוּץ מִמֶּנִּי׃ 26 כִּי לְאָדָם שֶׁטֹּוב לְפָנָיו נָתַן חָכְמָה וְדַעַת וְשִׂמְחָה וְלַחֹוטֶא נָתַן עִנְיָן לֶאֱסֹוף וְלִכְנֹוס לָתֵת לְטֹוב לִפְנֵי הָאֱלֹהִים גַּמ־זֶה הֶבֶל וּרְעוּת רוּחַ׃
