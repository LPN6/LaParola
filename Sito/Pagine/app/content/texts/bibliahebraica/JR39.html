1 בַּשָּׁנָה הַתְּשִׁעִית לְצִדְקִיָּהוּ מֶלֶכְ־יְהוּדָה בַּחֹדֶשׁ הָעֲשִׂרִי בָּא נְבוּכַדְרֶאצַּר מֶלֶכְ־בָּבֶל וְכָל־חֵילֹו אֶל־יְרוּשָׁלִַם וַיָּצֻרוּ עָלֶיהָ׃ ס 2 בְּעַשְׁתֵּי־עֶשְׂרֵה שָׁנָה לְצִדְקִיָּהוּ בַּחֹדֶשׁ הָרְבִיעִי בְּתִשְׁעָה לַחֹדֶשׁ הָבְקְעָה הָעִיר׃ 3 וַיָּבֹאוּ כֹּל שָׂרֵי מֶלֶכְ־בָּבֶל וַיֵּשְׁבוּ בְּשַׁעַר הַתָּוֶךְ נֵרְגַל שַׂר־אֶצֶר סַמְגַּר־נְבוּ שַׂר־סְכִים רַב־סָרִיס נֵרְגַל שַׂר־אֶצֶר רַב־מָג וְכָל־שְׁאֵרִית שָׂרֵי מֶלֶךְ בָּבֶל׃ 4 וַיְהִי כַּאֲשֶׁר רָאָם צִדְקִיָּהוּ מֶלֶכְ־יְהוּדָה וְכֹל ׀ אַנְשֵׁי הַמִּלְחָמָה וַיִּבְרְחוּ וַיֵּצְאוּ לַיְלָה מִנ־הָעִיר דֶּרֶךְ גַּן הַמֶּלֶךְ בְּשַׁעַר בֵּין הַחֹמֹתָיִם וַיֵּצֵא דֶּרֶךְ הָעֲרָבָה׃ 5 וַיִּרְדְּפוּ חֵיל־כַּשְׂדִּים אַחֲרֵיהֶם וַיַּשִּׂגוּ אֶת־צִדְקִיָּהוּ בְּעַרְבֹות יְרֵחֹו וַיִּקְחוּ אֹתֹו וַיַּעֲלֻהוּ אֶל־נְבוּכַדְרֶאצַּר מֶלֶכְ־בָּבֶל רִבְלָתָה בְּאֶרֶץ חֲמָת וַיְדַבֵּר אִתֹּו מִשְׁפָּטִים׃ 6 וַיִּשְׁחַט מֶלֶךְ בָּבֶל אֶת־בְּנֵי צִדְקִיָּהוּ בְּרִבְלָה לְעֵינָיו וְאֵת כָּל־חֹרֵי יְהוּדָה שָׁחַט מֶלֶךְ בָּבֶל׃ 7 וְאֶת־עֵינֵי צִדְקִיָּהוּ עִוֵּר וַיַּאַסְרֵהוּ בַּנְחֻשְׁתַּיִם לָבִיא אֹתֹו בָּבֶלָה׃ 8 וְאֶת־בֵּית הַמֶּלֶךְ וְאֶת־בֵּית הָעָם שָׂרְפוּ הַכַּשְׂדִּים בָּאֵשׁ וְאֶת־חֹמֹות יְרוּשָׁלִַם נָתָצוּ׃ 9 וְאֵת יֶתֶר הָעָם הַנִּשְׁאָרִים בָּעִיר וְאֶת־הַנֹּפְלִים אֲשֶׁר נָפְלוּ עָלָיו וְאֵת יֶתֶר הָעָם הַנִּשְׁאָרִים הֶגְלָה נְבוּזַר־אֲדָן רַב־טַבָּחִים בָּבֶל׃ 10 וּמִנ־הָעָם הַדַּלִּים אֲשֶׁר אֵינ־לָהֶם מְאוּמָה הִשְׁאִיר נְבוּזַרְאֲדָן רַב־טַבָּחִים בְּאֶרֶץ יְהוּדָה וַיִּתֵּן לָהֶם כְּרָמִים וִיגֵבִים בַּיֹּום הַהוּא׃ 11 וַיְצַו נְבוּכַדְרֶאצַּר מֶלֶכְ־בָּבֶל עַל־יִרְמְיָהוּ בְּיַד נְבוּזַרְאֲדָן רַב־טַבָּחִים לֵאמֹר׃ 12 קָחֶנּוּ וְעֵינֶיךָ שִׂים עָלָיו וְאַל־תַּעַשׂ לֹו מְאוּמָה רָּע כִּי אִם זז כַּאֲשֶׁר יְדַבֵּר אֵלֶיךָ כֵּן עֲשֵׂה עִמֹּו׃ 13 וַיִּשְׁלַח נְבוּזַרְאֲדָן רַב־טַבָּחִים וּנְבוּשַׁזְבָּן רַב־סָרִיס וְנֵרְגַל שַׂר־אֶצֶר רַב־מָג וְכֹל רַבֵּי מֶלֶכְ־בָּבֶל׃ 14 וַיִּשְׁלְחוּ וַיִּקְחוּ אֶת־יִרְמְיָהוּ מֵחֲצַר הַמַּטָּרָה וַיִּתְּנוּ אֹתֹו אֶל־גְּדַלְיָהוּ בֶּנ־אֲחִיקָם בֶּנ־שָׁפָן לְהֹוצִאֵהוּ אֶל־הַבָּיִת וַיֵּשֶׁב בְּתֹוךְ הָעָם׃ ס 15 וְאֶל־יִרְמְיָהוּ הָיָה דְבַר־יְהוָה בִּהְיֹתֹו עָצוּר בַּחֲצַר הַמַּטָּרָה לֵאמֹר׃ 16 הָלֹוךְ וְאָמַרְתָּ לְעֶבֶד־מֶלֶךְ הַכּוּשִׁי לֵאמֹר כֹּה־אָמַר יְהוָה צְבָאֹות אֱלֹהֵי יִשְׂרָאֵל הִנְנִי מֵבִי מֵבִיא אֶת־דְּבָרַי אֶל־הָעִיר הַזֹּאת לְרָעָה וְלֹא לְטֹובָה וְהָיוּ לְפָנֶיךָ בַּיֹּום הַהוּא׃ 17 וְהִצַּלְתִּיךָ בַיֹּומ־הַהוּא נְאֻמ־יְהוָה וְלֹא תִנָּתֵן בְּיַד הָאֲנָשִׁים אֲשֶׁר־אַתָּה יָגֹור מִפְּנֵיהֶם׃ 18 כִּי מַלֵּט אֲמַלֶּטְךָ וּבַחֶרֶב לֹא תִפֹּל וְהָיְתָה
