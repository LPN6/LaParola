1 וַיְדַבֵּר יְהוָה אֶל־מֹשֶׁה וְאֶל־אַהֲרֹן לֵאמֹר׃ 2 אִישׁ עַל־דִּגְלֹו בְאֹתֹת לְבֵית אֲבֹתָם יַחֲנוּ בְּנֵי יִשְׂרָאֵל מִנֶּגֶד סָבִיב לְאֹהֶל־מֹועֵד יַחֲנוּ׃ 3 וְהַחֹנִים קֵדְמָה מִזְרָחָה דֶּגֶל מַחֲנֵה יְהוּדָה לְצִבְאֹתָם וְנָשִׂיא לִבְנֵי יְהוּדָה נַחְשֹׁון בֶּנ־עַמִּינָדָב׃ 4 וּצְבָאֹו וּפְקֻדֵיהֶם אַרְבָּעָה וְשִׁבְעִים אֶלֶף וְשֵׁשׁ מֵאֹות׃ 5 וְהַחֹנִים עָלָיו מַטֵּה יִשָּׂשכָר וְנָשִׂיא לִבְנֵי יִשָּׂשכָר נְתַנְאֵל בֶּנ־צוּעָר׃ 6 וּצְבָאֹו וּפְקֻדָיו אַרְבָּעָה וַחֲמִשִּׁים אֶלֶף וְאַרְבַּע מֵאֹות׃ ס 7 מַטֵּה זְבוּלֻן וְנָשִׂיא לִבְנֵי זְבוּלֻן אֱלִיאָב בֶּנ־חֵלֹן׃ 8 וּצְבָאֹו וּפְקֻדָיו שִׁבְעָה וַחֲמִשִּׁים אֶלֶף וְאַרְבַּע מֵאֹות׃ 9 כָּל־הַפְּקֻדִים לְמַחֲנֵה יְהוּדָה מְאַת אֶלֶף וּשְׁמֹנִים אֶלֶף וְשֵׁשֶׁת־אֲלָפִים וְאַרְבַּע־מֵאֹות לְצִבְאֹתָם רִאשֹׁנָה יִסָּעוּ׃ ס 10 דֶּגֶל מַחֲנֵה רְאוּבֵן תֵּימָנָה לְצִבְאֹתָם וְנָשִׂיא לִבְנֵי רְאוּבֵן אֱלִיצוּר בֶּנ־שְׁדֵיאוּר׃ 11 וּצְבָאֹו וּפְקֻדָיו שִׁשָּׁה וְאַרְבָּעִים אֶלֶף וַחֲמֵשׁ מֵאֹות׃ 12 וְהַחֹונִם עָלָיו מַטֵּה שִׁמְעֹון וְנָשִׂיא לִבְנֵי שִׁמְעֹון שְׁלֻמִיאֵל בֶּנ־צוּרִי־שַׁדָּי׃ 13 וּצְבָאֹו וּפְקֻדֵיהֶם תִּשְׁעָה וַחֲמִשִּׁים אֶלֶף וּשְׁלֹשׁ מֵאֹות׃ 14 וְמַטֵּה גָּד וְנָשִׂיא לִבְנֵי גָד אֶלְיָסָף בֶּנ־רְעוּאֵל׃ 15 וּצְבָאֹו וּפְקֻדֵיהֶם חֲמִשָּׁה וְאַרְבָּעִים אֶלֶף וְשֵׁשׁ מֵאֹות וַחֲמִשִּׁים׃ 16 כָּל־הַפְּקֻדִים לְמַחֲנֵה רְאוּבֵן מְאַת אֶלֶף וְאֶחָד וַחֲמִשִּׁים אֶלֶף וְאַרְבַּע־מֵאֹות וַחֲמִשִּׁים לְצִבְאֹתָם וּשְׁנִיִּם יִסָּעוּ׃ ס 17 וְנָסַע אֹהֶל־מֹועֵד מַחֲנֵה הַלְוִיִּם בְּתֹוךְ הַמַּחֲנֹת כַּאֲשֶׁר יַחֲנוּ כֵּן יִסָּעוּ אִישׁ עַל־יָדֹו לְדִגְלֵיהֶם׃ ס 18 דֶּגֶל מַחֲנֵה אֶפְרַיִם לְצִבְאֹתָם יָמָּה וְנָשִׂיא לִבְנֵי אֶפְרַיִם אֱלִישָׁמָע בֶּנ־עַמִּיהוּד׃ 19 וּצְבָאֹו וּפְקֻדֵיהֶם אַרְבָּעִים אֶלֶף וַחֲמֵשׁ מֵאֹות׃ 20 וְעָלָיו מַטֵּה מְנַשֶּׁה וְנָשִׂיא לִבְנֵי מְנַשֶּׁה גַּמְלִיאֵל בֶּנ־פְּדָהצוּר׃ 21 וּצְבָאֹו וּפְקֻדֵיהֶם שְׁנַיִם וּשְׁלֹשִׁים אֶלֶף וּמָאתָיִם׃ 22 וּמַטֵּה בִּנְיָמִן וְנָשִׂיא לִבְנֵי בִנְיָמִן אֲבִידָן בֶּנ־גִּדְעֹנִי׃ 23 וּצְבָאֹו וּפְקֻדֵיהֶם חֲמִשָּׁה וּשְׁלֹשִׁים אֶלֶף וְאַרְבַּע מֵאֹות׃ 24 כָּל־הַפְּקֻדִים לְמַחֲנֵה אֶפְרַיִם מְאַת אֶלֶף וּשְׁמֹנַת־אֲלָפִים וּמֵאָה לְצִבְאֹתָם וּשְׁלִשִׁים יִסָּעוּ׃ ס 25 דֶּגֶל מַחֲנֵה דָן צָפֹנָה לְצִבְאֹתָם וְנָשִׂיא לִבְנֵי דָן אֲחִיעֶזֶר בֶּנ־עַמִּישַׁדָּי׃ 26 וּצְבָאֹו וּפְקֻדֵיהֶם שְׁנַיִם וְשִׁשִּׁים אֶלֶף וּשְׁבַע מֵאֹות׃ 27 וְהַחֹנִים עָלָיו מַטֵּה אָשֵׁר וְנָשִׂיא לִבְנֵי אָשֵׁר פַּגְעִיאֵל בֶּנ־עָכְרָן׃ 28 וּצְבָאֹו וּפְקֻדֵיהֶם אֶחָד וְאַרְבָּעִים אֶלֶף וַחֲמֵשׁ מֵאֹות׃ 29 וּמַטֵּה נַפְתָּלִי וְנָשִׂיא לִבְנֵי נַפְתָּלִי אֲחִירַע בֶּנ־עֵינָן׃ 30 וּצְבָאֹו וּפְקֻדֵיהֶם שְׁלֹשָׁה וַחֲמִשִּׁים אֶלֶף וְאַרְבַּע מֵאֹות׃ 31 כָּל־הַפְּקֻדִים לְמַחֲנֵה דָן מְאַת אֶלֶף וְשִׁבְעָה וַחֲמִשִּׁים אֶלֶף וְשֵׁשׁ מֵאֹות לָאַחֲרֹנָה יִסְעוּ לְדִגְלֵיהֶם׃ ף 32 אֵלֶּה פְּקוּדֵי בְנֵי־יִשְׂרָאֵל לְבֵית אֲבֹתָם כָּל־פְּקוּדֵי הַמַּחֲנֹת לְצִבְאֹתָם שֵׁשׁ־מֵאֹות אֶלֶף וּשְׁלֹשֶׁת אֲלָפִים וַחֲמֵשׁ מֵאֹות וַחֲמִשִּׁים׃ 33 וְהַלְוִיִּם לֹא הָתְפָּקְדוּ בְּתֹוךְ בְּנֵי יִשְׂרָאֵל כַּאֲשֶׁר צִוָּה יְהוָה אֶת־מֹשֶׁה׃ 34 וַיַּעֲשׂוּ בְּנֵי יִשְׂרָאֵל כְּכֹל אֲשֶׁר־צִוָּה יְהוָה אֶת־מֹשֶׁה כֵּנ־חָנוּ לְדִגְלֵיהֶם וְכֵן נָסָעוּ אִישׁ לְמִשְׁפְּחֹתָיו עַל־בֵּית אֲבֹתָיו׃
